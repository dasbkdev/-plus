### 3.1 Виртуальное сложение массивов

Условие задачи. По корректировке создать Array1 и Array2 с двумя закрытыми массивами unsigned и количеством элементов. Базовый виртуальный метод складывает все пары, переопределенный метод Array2 учитывает только четные значения каждой пары. Дополнительный Array3 складывает пары на нечетных позициях, сохраняя требование исходной методички о двух производных классах.

Данные программы. Array1: a[100], b[100], count; MAX — статическая константа. Array2: собственные a[100], b[100], count по условию корректировки. Array3 использует защищенные методы чтения базового класса.

Проверки и ограничения. Размер: 1..100; элементы при вводе: 0..2147483647. На проверенной платформе unsigned 32-битный; максимальная вводимая сумма 4294967294 помещается в тип. Защитная проверка остается в printSum(). «Четный элемент» означает четность значения, а не индекса. В Array3 позиции считаются с единицы.

#### 3.1.1 Алгоритм

1. Задать размер и заполнить два массива неотрицательными целыми числами.

2. Вызвать sum() напрямую для Array1, Array2 и Array3; перед сложением проверить предел unsigned.

3. Показать виртуальные вызовы через Array1* и Array1&.

4. Вызвать even.Array1::sum(), явно выбирая базовую реализацию.

5. Отдельно проверить конструкторы со значением по умолчанию 0.

Входные данные. Пункт меню; длина и элементы двух массивов.

Выходные данные. Суммы пар по правилам Array1, Array2 и Array3; результаты виртуальных и явных вызовов.

```cpp
#include "../common/input.h"
#include <limits>

// Три класса по-разному складывают одни и те же два массива.
// Виртуальный метод выбирается по настоящему типу объекта.
// Массивы обычные: не более 100 чисел в каждом.
class Array1 {
  public:
    static const int MAX = 100;

  private:
    unsigned a[MAX], b[MAX];
    int count;

  protected:
    unsigned first(int i) const { return a[i]; }
    unsigned second(int i) const { return b[i]; }
    int size() const { return count; }
    void printSum(unsigned x, unsigned y) const {
        // До сложения проверяем, поместится ли результат в unsigned.
        if (x > numeric_limits<unsigned>::max() - y) cout << "Переполнение ";
        else cout << x + y << ' ';
    }

  public:
    Array1(int n = 1, unsigned value = 0) : count(n) {
        for (int i = 0; i < count; i++) a[i] = b[i] = value;
    }
    Array1(int n, const unsigned x[], const unsigned y[]) : count(n) {
        for (int i = 0; i < count; i++) {
            a[i] = x[i];
            b[i] = y[i];
        }
    }
    virtual ~Array1() {}
    virtual void sum() const {
        cout << "Все пары: ";
        for (int i = 0; i < count; i++) printSum(a[i], b[i]);
        cout << '\n';
    }
};

class Array2 : public Array1 {
  private:
    unsigned a[MAX], b[MAX];
    int count;

  public:
    Array2(int n = 1, unsigned value = 0) : Array1(n, value), count(n) {
        for (int i = 0; i < count; i++) a[i] = b[i] = value;
    }
    Array2(int n, const unsigned x[], const unsigned y[])
        : Array1(n, x, y), count(n) {
        for (int i = 0; i < count; i++) {
            a[i] = x[i];
            b[i] = y[i];
        }
    }
    ~Array2() override {}
    // Переопределение сохраняет сигнатуру и меняет действие: берем четные
    // значения.
    void sum() const override {
        cout << "Суммы только четных значений в каждой паре: ";
        for (int i = 0; i < count; i++) {
            unsigned firstEven = 0;
            unsigned secondEven = 0;
            // Нечетное значение не участвует в сумме этой пары.
            if (a[i] % 2 == 0) firstEven = a[i];
            if (b[i] % 2 == 0) secondEven = b[i];
            printSum(firstEven, secondEven);
        }
        cout << '\n';
    }
};

class Array3 : public Array1 {
  public:
    Array3(int n, const unsigned x[], const unsigned y[]) : Array1(n, x, y) {}
    void sum() const override {
        cout << "Пары на нечетных позициях (1, 3, ...): ";
        // Индексы 0, 2, 4 соответствуют позициям 1, 3, 5.
        for (int i = 0; i < size(); i += 2) printSum(first(i), second(i));
        cout << '\n';
    }
};

int main() {
    while (true) {
        cout << "\n1. Ввести массивы и показать все вызовы\n2. Конструкторы с "
                "нулями\n0. Выход\n";
        int menu = readInt("Выбор: ", 0, 2);
        if (menu == 0) break;
        if (menu == 2) {
            Array1 a;
            Array2 b;
            a.sum();
            b.sum();
            continue;
        }
        int n = readInt("Размер (1..100): ", 1, Array1::MAX);
        unsigned x[Array1::MAX], y[Array1::MAX];
        cout << "Первый массив:\n";
        for (int i = 0; i < n; i++) x[i] = readInt("Элемент: ", 0, 2147483647);
        cout << "Второй массив:\n";
        for (int i = 0; i < n; i++) y[i] = readInt("Элемент: ", 0, 2147483647);
        Array1 base(n, x, y);
        Array2 even(n, x, y);
        Array3 oddPositions(n, x, y);
        cout << "Прямые вызовы:\n";
        base.sum();
        even.sum();
        oddPositions.sum();
        // Указатель хранит адрес; & берет адрес существующего объекта.
        Array1 *pointer = &even;
        cout << "Через указатель базового класса:\n";
        pointer->sum(); // Выбирается метод реального объекта Array2.
        Array1 &reference = oddPositions;
        // Ссылка — другое имя того же объекта, копия не создается.
        cout << "Через ссылку базового класса:\n";
        reference.sum();
        cout << "Явный вызов базового метода для Array2:\n";
        even.Array1::sum();
    }
}
```

### 3.2 Площади фигур через абстрактный класс

Условие задачи. Выполнить корректировку 2026 года: абстрактный Figure хранит линейный размер R и имеет чисто виртуальный area(). Создать Circle, Square и Triangle для круга, квадрата и равностороннего треугольника. Использовать глобальную константу pi = 3.14; исходный вариант прямоугольника и трапеции заменен корректировкой.

Данные программы. Figure: защищенное double R, конструктор со значением 1. Circle, Square и Triangle переопределяют area(); showArea принимает const Figure&.

Проверки и ограничения. Размер: 0.000001..1000000000. Pi = 3.14 взято именно из задания. Обычные площади выводятся с шестью знаками после точки, очень маленькие — в экспоненциальном виде, чтобы не превращать положительную площадь в видимый ноль.

#### 3.2.1 Алгоритм

1. Выбрать фигуру и ввести положительный радиус или сторону.

2. Вычислить площадь: 3.14*R*R, R*R или sqrt(3)*R*R/4.

3. Передать объект общей функции showArea(); виртуально выбрать нужную формулу.

4. Проверить отдельным пунктом все фигуры с R = 1.

Входные данные. Номер пункта меню и значения, которые запрашивает выбранная операция. Figure: защищенное double R, конструктор со значением 1. Circle, Square и Triangle переопределяют area(); showArea принимает const Figure&.

Выходные данные. Результат выбранного действия, текущее состояние данных или точное сообщение об ошибке. Вызвать area через Figure: круг pi*R*R, квадрат R*R, треугольник sqrt(3)*R*R/4.

#### 3.2.2 Блок-схема

#### 3.2.3 Программный код

```cpp
#include "../common/input.h"
#include <cmath>
#include <iomanip>

// Общий класс Figure задает обязательный метод вычисления площади.
// Круг, квадрат и треугольник подставляют в него свою формулу.
// Одна функция showArea умеет вывести площадь любой из этих фигур.
const double pi = 3.14;

class Figure {
  protected:
    double R;

  public:
    Figure() : R(1) {}
    explicit Figure(double r) : R(r) {}
    virtual ~Figure() {
    } // При удалении через родителя вызывается нужный деструктор.
    virtual double
    area() const = 0; // Абстрактный класс задает общий интерфейс.
};

class Circle : public Figure {
  public:
    Circle(double r = 1) : Figure(r) {}
    double area() const override {
        // override подтверждает, что мы меняем виртуальный метод родителя.
        return pi * R * R;
    }
};

class Square : public Figure {
  public:
    Square(double r = 1) : Figure(r) {}
    double area() const override { return R * R; }
};

class Triangle : public Figure {
  public:
    Triangle(double r = 1) : Figure(r) {}
    double area() const override { return sqrt(3.0) * R * R / 4; }
};

void showArea(const Figure &figure) {
    // const Figure& передает объект без копирования и запрещает менять его.
    double area = figure.area(); // Один виртуальный вызов для любой фигуры.
    if (area < 0.000001) cout << scientific;
    cout << "Площадь: " << area << '\n';
    cout << fixed;
}

int main() {
    cout << fixed << setprecision(6);
    while (true) {
        cout << "\n1. Круг\n2. Квадрат\n3. Равносторонний треугольник\n"
             << "4. Все фигуры с R = 1\n0. Выход\n";
        int menu = readInt("Выбор: ", 0, 4);
        if (menu == 0) break;
        if (menu == 4) {
            Circle circle;
            Square square;
            Triangle triangle;
            cout << "Круг. ";
            showArea(circle);
            cout << "Квадрат. ";
            showArea(square);
            cout << "Треугольник. ";
            showArea(triangle);
            continue;
        }
        string prompt = "Сторона: ";
        if (menu == 1) prompt = "Радиус: ";
        double r = readDouble(prompt, 0.000001, 1e9);
        if (menu == 1) {
            Circle figure(r);
            showArea(figure);
        }
        if (menu == 2) {
            Square figure(r);
            showArea(figure);
        }
        if (menu == 3) {
            Triangle figure(r);
            showArea(figure);
        }
    }
}
```

### 3.3 Перевод денежных сумм

Условие задачи. Создать абстрактный Currency с чисто виртуальным conv() и виртуальным showConv(). Классы Dollar и Euro реализуют перевод в рубли и обратно. По корректировке заданы условные курсы 74 и 99 рублей.

Данные программы. Currency: amount, toRubles. Dollar и Euro определяют курс, conv() и формат вывода showConv().

Проверки и ограничения. Сумма: 0..1000000000. Курсы учебные и фиксированные; сетевые курсы не используются. Для вывода сумма округляется до двух знаков после точки.

#### 3.3.1 Алгоритм

1. Выбрать валюту и направление перевода.

2. Прочитать неотрицательную сумму.

3. Для перевода в рубли умножить сумму на курс, для обратного перевода разделить на курс.

4. Вызвать виртуальный showConv() через ссылку Currency.

Входные данные. Номер пункта меню и значения, которые запрашивает выбранная операция. Currency: amount, toRubles. Dollar и Euro определяют курс, conv() и формат вывода showConv().

Выходные данные. Результат выбранного действия, текущее состояние данных или точное сообщение об ошибке. Через Currency перевести сумму: умножить на курс 74 или 99 либо разделить.

#### 3.3.2 Блок-схема

#### 3.3.3 Программный код

```cpp
#include "../common/input.h"
#include <iomanip>

// Программа переводит доллары и евро в рубли и обратно.
// Currency хранит общие данные, а потомки задают разные курсы.
// Курсы 74 и 99 взяты из задания и не меняются через интернет.
class Currency {
  protected:
    double amount;
    bool toRubles;

  public:
    Currency(double a, bool to) : amount(a), toRubles(to) {}
    virtual ~Currency() {}
    virtual double conv() const = 0;
    virtual void showConv() const { cout << conv() << '\n'; }
};

class Dollar : public Currency {
  public:
    Dollar(double a, bool to) : Currency(a, to) {}
    double conv() const override {
        if (toRubles) return amount * 74;
        return amount / 74;
    }
    void showConv() const override {
        if (toRubles) cout << amount << " USD = " << conv() << " RUB\n";
        else cout << amount << " RUB = " << conv() << " USD\n";
    }
};

class Euro : public Currency {
  public:
    Euro(double a, bool to) : Currency(a, to) {}
    double conv() const override {
        if (toRubles) return amount * 99;
        return amount / 99;
    }
    void showConv() const override {
        if (toRubles) cout << amount << " EUR = " << conv() << " RUB\n";
        else cout << amount << " RUB = " << conv() << " EUR\n";
    }
};

void show(const Currency &currency) {
    // Ссылка на родителя позволяет передать доллар или евро в одну функцию.
    currency.showConv(); // Виртуальный метод выбирается по типу валюты.
}

int main() {
    cout << fixed << setprecision(2);
    cout << "Учебные курсы: 1 USD = 74 RUB, 1 EUR = 99 RUB.\n";
    while (true) {
        cout << "\n1. Доллары в рубли\n2. Рубли в доллары\n3. Евро в рубли\n4. "
                "Рубли в евро\n0. Выход\n";
        int menu = readInt("Выбор: ", 0, 4);
        if (menu == 0) break;
```

```cpp
inline string readLine(string prompt) {
    cout << prompt;
    string line;
    if (!getline(cin, line)) {
        cout << "\nВвод завершён.\n";
        exit(0);
    }
    return line;
}

inline string readText(string prompt, size_t maxLength = 100) {
    while (true) {
        string text = readLine(prompt);
        if (text.size() > maxLength) {
            cout << "Ошибка: введите непустой текст (до " << maxLength
                 << " байт UTF-8).\n";
            continue;
        }
        // Убираем пробелы в начале и конце ответа.
        while (!text.empty() && (text.front() == ' ' || text.front() == '\t' ||
                                 text.front() == '\r')) {
            text.erase(0, 1);
        }
        while (!text.empty() && (text.back() == ' ' || text.back() == '\t' ||
                                 text.back() == '\r')) {
            text.pop_back();
        }

        bool valid = true;
        if (text.empty()) {
            valid = false;
        }
        for (unsigned char c : text) {
            // Служебные символы не должны ломать таблицу в консоли.
            if (c < 32 || c == 127) {
                valid = false;
            }
        }
        if (valid) {
            return text;
        }
        cout << "Ошибка: введите непустой текст (до " << maxLength
             << " байт UTF-8).\n";
    }
}

inline int readInt(string prompt, int low, int high) {
    while (true) {
        string line = readLine(prompt);
        // Читаем число из всей строки, чтобы не принять 12abc за число 12.
        istringstream input(line);
        int value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        if (value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно целое число от " << low << " до " << high
             << ".\n";
    }
}

inline double readDouble(string prompt, double low, double high) {
    while (true) {
        istringstream input(readLine(prompt));
        double value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        // isfinite отсеивает бесконечность и недопустимый результат NaN.
        if (!isfinite(value) || value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно конечное число от " << low << " до " << high
             << ". Дробная часть отделяется точкой.\n";
    }
}

inline bool validDate(string date) {
    if (date.size() != 10 || date[2] != '.' || date[5] != '.') {
        return false;
    }
    for (int i = 0; i < 10; i++) {
        if (i != 2 && i != 5 && (date[i] < '0' || date[i] > '9')) {
            return false;
        }
    }
    // substr берёт часть строки, stoi переводит её в целое число.
    int day = stoi(date.substr(0, 2));
    int month = stoi(date.substr(3, 2));
    int year = stoi(date.substr(6, 4));
    if (year < 1900 || year > 2100 || month < 1 || month > 12) {
        return false;
    }
    int days[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    // Високосный год добавляет один день в феврале.
    if (year % 400 == 0 || (year % 4 == 0 && year % 100 != 0)) {
        days[1] = 29;
    }
    return day >= 1 && day <= days[month - 1];
}

           stoi(date.substr(0, 2));
}

inline string readTime(string prompt) {
    while (true) {
        string time = readText(prompt, 5);
        bool valid = time.size() == 5;
        if (valid) {
            valid = time[2] == ':';
            for (int i = 0; i < 5; i++) {
                if (i != 2 && (time[i] < '0' || time[i] > '9')) {
                    valid = false;
                }
            }
        }
        if (valid) {
            int hours = stoi(time.substr(0, 2));

                return time;
            }
        }

    }
}
```
