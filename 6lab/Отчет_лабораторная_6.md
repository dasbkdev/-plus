### 3.1 Работник и менеджер

Условие задачи. Создать Employee с именем, должностью и месячным окладом. Вычислять годовую зарплату. Создать Manager с бонусом, машиной компании и опционами. По корректировке 2026 года использовать protected-наследование, ввод данных, отдельный вывод собственных полей менеджера и всех полей, открытые методы для доступа к родителю.

Данные программы. Employee: name, position, salary — закрытые данные работника. Manager: annual_bonus, company_car, stock_options — закрытые данные менеджера. showEmployee() и employeeYear() — открытые методы-обертки над защищенно унаследованными методами.

Проверки и ограничения. Оклад и годовой бонус: 0..1000000000; опционы: 0..1000000. Текстовые поля: 1..63 байта UTF-8. Объект хранит обычные string вместо небезопасных символьных буферов. Поля остаются закрытыми.

#### 3.1.1 Алгоритм

1. Создать начальные объекты работника и менеджера конструкторами по умолчанию.

2. Прочитать поля через input(); годовой оклад вычислить как salary * 12.

3. По меню вывести работника, только собственные поля менеджера или полную информацию.

4. К годовому окладу менеджера прибавить годовой бонус; опционы считать количеством, их денежная цена не задана.

5. Проверить методы-обертки, конструкторы по умолчанию и корректно завершить программу.

Входные данные. Пункт меню; имя, должность, оклад, бонус, машина и число опционов.

Выходные данные. Данные работника и менеджера; годовая зарплата с учётом бонуса; сообщения об ошибках.

```cpp
#include "../common/input.h"
#include <iomanip>

// Программа хранит данные работника и менеджера.
// Менеджер получает общие методы работника через наследование.
// Проверки обычного ввода вынесены в общий файл input.h.
class Employee {
  private:
    string name;
    string position;
    double salary;

  public:
    // После двоеточия задаются начальные значения полей объекта.
    Employee() : name("Не задано"), position("Не задано"), salary(0) {}
    Employee(string n, string p, double s) : name(n), position(p), salary(s) {}
    ~Employee() {}
    void input() {
        name = readText("Имя: ", 63);
        position = readText("Должность: ", 63);
        salary = readDouble("Оклад в месяц (0..1000000000): ", 0, 1e9);
    }
    double annualSalary() const {
        // В году 12 месяцев. const означает, что поля не изменяются.
        return salary * 12;
    }
    void show() const {
        cout << "Работник: " << name << "\nДолжность: " << position
             << "\nОклад за месяц: " << salary
             << "\nОклад за год: " << annualSalary() << '\n';
    }
};

// По корректировке методички используется защищенное наследование.
class Manager : protected Employee {
  private:
    double annual_bonus;
    string company_car;
    int stock_options;

  public:
    Manager() : annual_bonus(0), company_car("Нет"), stock_options(0) {}
    Manager(string n, string p, double s, double b, string car, int options)
        : Employee(n, p, s), annual_bonus(b), company_car(car),
          stock_options(options) {}
    ~Manager() {}
    void input() {
        // Сначала вводятся общие данные, затем собственные данные менеджера.
        Employee::input();
        annual_bonus = readDouble("Бонус в год: ", 0, 1e9);
        company_car = readText("Машина компании: ", 63);
        stock_options = readInt("Число опционов: ", 0, 1000000);
    }
    // Открытые обертки разрешают обратиться к защищенным методам родителя.
    void showEmployee() const { Employee::show(); }
    double employeeYear() const { return annualSalary(); }
    void showOwn() const {
        cout << "Бонус за год: " << annual_bonus << "\nМашина: " << company_car
             << "\nОпционы: " << stock_options << '\n';
    }
    void show() const {
        Employee::show(); // Закрытые поля доступны через методы родителя.
        showOwn();
        cout << "Доход за год с бонусом: " << annualSalary() + annual_bonus
             << '\n';
    }
};

int main() {
    Employee employee;
    Manager manager;
    cout << fixed << setprecision(2);
    while (true) {
        // После каждого действия снова показываем меню.
        cout << "\n1. Создать работника\n2. Показать работника\n3. Создать "
                "менеджера\n"
             << "4. Только поля менеджера\n5. Все данные менеджера\n6. Объекты "
                "по умолчанию\n"
             << "7. Данные родителя через обертку\n8. Годовой оклад через "
                "обертку\n0. Выход\n";
        int menu = readInt("Выбор: ", 0, 8);
        if (menu == 0) break;
        if (menu == 1) employee.input();
        if (menu == 3) manager.input();
        if (menu == 2) employee.show();
        if (menu == 4) manager.showOwn();
        if (menu == 5) manager.show();
        if (menu == 7) manager.showEmployee();
        if (menu == 8)
            cout << "Годовой оклад: " << manager.employeeYear() << '\n';
        if (menu == 6) {
            Employee defaultEmployee;
            Manager defaultManager;
            defaultEmployee.show();
            defaultManager.show();
        }
    }
}
```

### 3.2 Дисплей, материнская плата и компьютер

Условие задачи. Разработать Display и MotherBoard. Создать Computer, добавляющий марку и размер диска. По корректировке использовать public Display и protected MotherBoard, показать собственные поля потомка и полную информацию, восстановить внешний доступ к сведениям платы методом-оберткой.

Данные программы. Display: type, colors, x_resolution, y_resolution. MotherBoard: processor (целочисленный код), speed (МГц), RAM (ГБ). Computer: name, hard_disk (ГБ); showMotherBoard() — открытая обертка.

Проверки и ограничения. Цвета: 2..1073741824; разрешение: 1..32768 по каждой оси. Код процессора: 1..1000000; частота: 1..100000 МГц; память: 1..1048576 ГБ. Объем диска: 1..1000000000 ГБ; текст 31 или 63 байта UTF-8.

#### 3.2.1 Алгоритм

1. Создать учебный компьютер с заданными исходными значениями.

2. Ввести данные, вызвать конструктор Computer и передать параметры конструкторам обоих родителей.

3. Метод Show() выводит собственные данные, затем монитор и плату.

4. Отдельными пунктами меню проверить доступ к монитору, обертку платы и самостоятельные объекты родителей.

Входные данные. Номер пункта меню и значения, которые запрашивает выбранная операция. Display: type, colors, x_resolution, y_resolution. MotherBoard: processor (целочисленный код), speed (МГц), RAM (ГБ). Computer: name, hard_disk (ГБ); showMotherBoard() — открытая обертка.

Выходные данные. Результат выбранного действия, текущее состояние данных или точное сообщение об ошибке. Собрать Computer из Display и MotherBoard; показать плату через открытый метод.

```cpp
#include "../common/input.h"

// Компьютер объединяет сведения о мониторе и материнской плате.
// Это пример наследования сразу от двух классов.
// Каждый пункт меню показывает отдельную часть данных.
class Display {
  private:
    string type;
    long colors;
    int x_resolution, y_resolution;

  public:
    Display(string t, long c, int x, int y)
        // Конструктор сразу заполняет закрытые поля монитора.
        : type(t), colors(c), x_resolution(x), y_resolution(y) {}
    ~Display() {}
    void showDisplay() const {
        cout << "Монитор: " << type << ", цветов: " << colors
             << ", разрешение: " << x_resolution << 'x' << y_resolution << '\n';
    }
};

class MotherBoard {
  private:
    int processor, speed, RAM;

  public:
    MotherBoard(int p, int s, int r) : processor(p), speed(s), RAM(r) {}
    ~MotherBoard() {}
    void showBoard() const {
        cout << "Код процессора: " << processor << ", частота (МГц): " << speed
             << ", ОЗУ (ГБ): " << RAM << '\n';
    }
};

// У компьютера два родителя с разными механизмами доступа.
class Computer : public Display, protected MotherBoard {
  private:
    string name;
    int hard_disk;

  public:
    Computer(string t, long c, int x, int y, int p, int s, int r, string n,
             int disk)
        // Сначала создаются оба родителя, затем поля самого компьютера.
        : Display(t, c, x, y), MotherBoard(p, s, r), name(n), hard_disk(disk) {}
    ~Computer() {}
    void showOwn() const {
        cout << "Марка: " << name << ", диск (ГБ): " << hard_disk << '\n';
    }
    void showMotherBoard() const {
        MotherBoard::showBoard(); // Открытый метод-обертка восстанавливает
                                  // доступ.
    }
    void Show() const {
        // Полная информация состоит из трех частей.
        showOwn();
        showDisplay();
        showMotherBoard();
    }
};

int main() {
    Computer computer("IPS", 16777216, 1920, 1080, 5, 3200, 16, "Учебный ПК",
                      512);
    while (true) {
        cout << "\n1. Ввести компьютер\n2. Поля самого компьютера\n3. Все "
                "данные\n"
             << "4. Монитор\n5. Плата через открытый метод\n6. Проверить "
                "отдельных родителей\n0. Выход\n";
        int menu = readInt("Выбор: ", 0, 6);
        if (menu == 0) break;
        if (menu == 1) {
            string type = readText("Тип монитора: ", 31);
            int colors = readInt("Цветов: ", 2, 1073741824);
            int x = readInt("Разрешение X: ", 1, 32768);
            int y = readInt("Разрешение Y: ", 1, 32768);
            int processor = readInt("Код процессора: ", 1, 1000000);
            int speed = readInt("Частота (МГц): ", 1, 100000);
            int ram = readInt("ОЗУ (ГБ): ", 1, 1048576);
            string name = readText("Марка: ", 63);
            int disk = readInt("Объем диска (ГБ): ", 1, 1000000000);
            computer =
                Computer(type, colors, x, y, processor, speed, ram, name, disk);
        }
        if (menu == 2) computer.showOwn();
        if (menu == 3) computer.Show();
        if (menu == 4) computer.showDisplay();
        if (menu == 5) computer.showMotherBoard();
        if (menu == 6) {
            // Родительские классы можно использовать и без компьютера.
            Display display("IPS", 16777216, 1920, 1080);
            MotherBoard board(5, 3200, 16);
            display.showDisplay();
            board.showBoard();
        }
    }
}
```

### 3.3 Восстановление доступа при наследовании

Условие задачи. Изменить механизм наследования в задаче о работнике и менеджере и показать способы восстановления доступа. Уточнение: private-наследование ограничивает внешний доступ к public/protected-членам; private-поля самого Employee недоступны потомку напрямую при любом механизме наследования.

Данные программы. Employee: name, position, salary. ManagerWrapper, ManagerUsing, ManagerPublic: annual_bonus, company_car, stock_options. Три варианта отличаются способом предоставления доступа к show() и annualSalary().

Проверки и ограничения. Оклад и бонус: 0..1000000000; опционы: 0..1000000. Закрытые поля не раскрываются: извне вызываются только методы. Некорректные обращения к закрытым членам объяснены, но не включены в компилируемый код.

3. Проверить третий вариант с public-наследованием.

Входные данные. Номер пункта меню и значения, которые запрашивает выбранная операция. Employee: name, position, salary. ManagerWrapper, ManagerUsing, ManagerPublic: annual_bonus, company_car, stock_options. Три варианта отличаются способом предоставления доступа к show() и annualSalary().

Файл common/input.h подключается строкой #include "../common/input.h". Он содержит общие функции чтения и проверки данных. Для сборки программы папку common необходимо сохранять рядом с папками лабораторных.

#ifndef INPUT_H
#define INPUT_H

#include <iostream>
#include <string>
#include <sstream>
#include <cmath>
#include <cstdlib>
#include <iomanip>
using namespace std;

// Здесь только ввод. Алгоритмы самих задач находятся в файлах taskN.cpp.
// inline позволяет хранить эти небольшие функции в общем заголовочном файле.
inline string readLine(string prompt) {
    cout << prompt;
    string line;
    if (!getline(cin, line)) {
        cout << "\nВвод завершён.\n";
        exit(0);
    }
    return line;
}

inline string readText(string prompt, size_t maxLength = 100) {
    while (true) {
        string text = readLine(prompt);
        if (text.size() > maxLength) {
            cout << "Ошибка: введите непустой текст (до " << maxLength
                 << " байт UTF-8).\n";
            continue;
        }
        // Убираем пробелы в начале и конце ответа.
        while (!text.empty() && (text.front() == ' ' || text.front() == '\t' ||
                                 text.front() == '\r')) {
            text.erase(0, 1);
        }
        while (!text.empty() && (text.back() == ' ' || text.back() == '\t' ||
                                 text.back() == '\r')) {
            text.pop_back();
        }

        bool valid = true;
        if (text.empty()) {
            valid = false;
        }
        for (unsigned char c : text) {
            // Служебные символы не должны ломать таблицу в консоли.
            if (c < 32 || c == 127) {
                valid = false;
            }
        }
        if (valid) {
            return text;
        }
        cout << "Ошибка: введите непустой текст (до " << maxLength
             << " байт UTF-8).\n";
    }
}

inline int readInt(string prompt, int low, int high) {
    while (true) {
        string line = readLine(prompt);
        // Читаем число из всей строки, чтобы не принять 12abc за число 12.
        istringstream input(line);
        int value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        if (value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно целое число от " << low << " до " << high
             << ".\n";
    }
}

inline double readDouble(string prompt, double low, double high) {
    while (true) {
        istringstream input(readLine(prompt));
        double value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        // isfinite отсеивает бесконечность и недопустимый результат NaN.
        if (!isfinite(value) || value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно конечное число от " << low << " до " << high
             << ". Дробная часть отделяется точкой.\n";
    }
}

inline bool validDate(string date) {
    if (date.size() != 10 || date[2] != '.' || date[5] != '.') {
        return false;
    }
    for (int i = 0; i < 10; i++) {
        if (i != 2 && i != 5 && (date[i] < '0' || date[i] > '9')) {
            return false;
        }
    }
    // substr берёт часть строки, stoi переводит её в целое число.
    int day = stoi(date.substr(0, 2));
    int month = stoi(date.substr(3, 2));
    int year = stoi(date.substr(6, 4));
    if (year < 1900 || year > 2100 || month < 1 || month > 12) {
        return false;
    }
    int days[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    // Високосный год добавляет один день в феврале.
    if (year % 400 == 0 || (year % 4 == 0 && year % 100 != 0)) {
        days[1] = 29;
    }
    return day >= 1 && day <= days[month - 1];
}

// Запись ГГГГММДД позволяет правильно сравнить две даты как числа.
inline int dateNumber(string date) {
    return stoi(date.substr(6, 4)) * 10000 + stoi(date.substr(3, 2)) * 100 +
           stoi(date.substr(0, 2));
}

inline string readDate(string prompt) {
    while (true) {
        string date = readText(prompt, 10);
        if (validDate(date)) {
            return date;
        }
        cout << "Ошибка: нужна существующая дата ДД.ММ.ГГГГ, год 1900..2100.\n";
    }
}

inline string readTime(string prompt) {
    while (true) {
        string time = readText(prompt, 5);
        bool valid = time.size() == 5;
        if (valid) {
            valid = time[2] == ':';
            for (int i = 0; i < 5; i++) {
                if (i != 2 && (time[i] < '0' || time[i] > '9')) {
                    valid = false;
                }
            }
        }
        if (valid) {
            int hours = stoi(time.substr(0, 2));
            int minutes = stoi(time.substr(3, 2));
            if (hours <= 23 && minutes <= 59) {
                return time;
            }
        }
        cout << "Ошибка: время должно быть от 00:00 до 23:59.\n";
    }
}

inline string readPhone(string prompt) {
    while (true) {
        string phone = readText(prompt, 16);
        size_t start = 0;
        if (phone[0] == '+') {
            start = 1;
        }
        bool valid = phone.size() - start >= 7 && phone.size() - start <= 15;
        for (size_t i = start; i < phone.size(); i++) {
            if (phone[i] < '0' || phone[i] > '9') {
                valid = false;
            }
        }
        if (valid) {
            return phone;
        }
        cout << "Ошибка: нужно 7..15 цифр; знак + допустим только в начале.\n";
    }
}
