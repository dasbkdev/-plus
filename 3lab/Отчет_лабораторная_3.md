### 3.1 Таймер обратного отсчёта

Условие задачи. Создать Timer с тремя конструкторами: секунды целым числом, секунды строкой через atoi, минуты и секунды двумя целыми. Метод run уменьшает время до нуля и выдаёт звуковой сигнал. Для отсчёта использовать clock.

Данные программы. seconds — оставшееся число секунд. Timer(int), Timer(const char*), Timer(int,int) задают одно и то же поле разными способами; run выполняет отсчёт.

Проверки и ограничения. Время 0..3600 секунд; минуты 0..60, секунды 0..59, при 60 минутах секунды только 0. До atoi проверяются цифры и длина строки, поэтому нет преобразования мусора или переполнения. CLOCKS_PER_SEC — стандартная современная замена устаревшего CLK_TCK. clock измеряет процессорное время: при нагрузке реальные секунды могут быть длиннее. Активное ожидание выбрано по требованию задания. Звук зависит от настроек терминала; сообщение о завершении выводится всегда.

#### 3.1.1 Алгоритм

1. Выбрать способ задания времени.

2. Проверить диапазон и формат данных.

3. Создать Timer соответствующим конструктором.

4. Пока seconds больше нуля, вывести значение и дождаться одной секунды по clock.

5. Уменьшить seconds и повторить цикл.

6. При нуле вывести сообщение и символ звукового сигнала.

#### 3.1.3 Программный код

```cpp
#include "../common/input.h"
#include <ctime>

// Три конструктора заполняют одно и то же поле seconds.
// Разница только в том, как пользователь задаёт время.
class Timer {
  private:
    int seconds;

  public:
    // Перегрузка позволяет задать время тремя способами.
    Timer(int value) : seconds(value) {}
    // atoi переводит заранее проверенную строку цифр в целое число.
    Timer(const char *value) : seconds(atoi(value)) {}
    Timer(int minutes, int value) : seconds(minutes * 60 + value) {}

    void run() {
        while (seconds > 0) {
            cout << seconds << '\n' << flush;
            clock_t start = clock();
            if (start == clock_t(-1)) {
                cout << "Таймер процессора недоступен.\n";
                return;
            }
            // По условию используем clock: отсчитывается процессорное время.
            while (true) {
                clock_t current = clock();
                if (current == clock_t(-1) || current < start) {
                    cout << "Ошибка часов процессора.\n";
                    return;
                }
                double passed = double(current - start) / CLOCKS_PER_SEC;
                if (passed >= 1.0) {
                    break;
                }
            }
            seconds--; // Одна секунда прошла, уменьшаем остаток.
        }
        cout << "0\nВремя вышло!\a\n";
    }
};

int main() {
    while (true) {
        cout << "\n1 - секунды числом, 2 - секунды строкой, 3 - минуты и "
                "секунды, 0 - выход\n";
        int menu = readInt("Выбор: ", 0, 3);
        if (menu == 0) {
            break;
        }
        if (menu == 1) {
            int seconds = readInt("Секунды (0..3600): ", 0, 3600);
            Timer timer(seconds);
            timer.run();
        }
        if (menu == 2) {
            string text;
            int value;
            bool valid;
            do {
                text = readText("Строка секунд (0..3600): ", 4);
                valid = true;
                // До atoi проверяем каждый символ, чтобы не принять текст за
                // ноль.
                for (char c : text) {
                    if (c < '0' || c > '9') {
                        valid = false;
                    }
                }
                value = 0;
                if (valid) {
                    value = atoi(text.c_str());
                }
                if (!valid || value > 3600) {
                    cout << "Ошибка: введите цифрами число от 0 до 3600.\n";
                }
            } while (!valid || value > 3600);
            Timer timer(
                text.c_str()); // c_str даёт строку в формате const char*.
            timer.run();
        }
        if (menu == 3) {
            int minutes = readInt("Минуты (0..60): ", 0, 60);
            int maxSeconds = 59;
            if (minutes == 60) {
                maxSeconds = 0; // Общее время не должно превышать час.
            }
            int seconds = readInt("Секунды: ", 0, maxSeconds);
            Timer timer(minutes, seconds);
            timer.run();
        }
    }
}
```

### 3.2 Товары на складе Nomenclature

Условие задачи. Создать класс товара с закрытыми названием, оптовой и розничной ценами и количеством. Использовать конструкторы и деструктор. Вычислять доход по товару и всему складу, показывать таблицу, сортировать и искать по всем полям через меню. Поле розничной цены и формула взяты из исправлений 2026 года.

Данные программы. name — название; wholesale и retail — цены; quantity — количество. Конструктор по умолчанию обнуляет числовые поля, конструктор с аргументами использует список инициализации. profit = (retail - wholesale) * quantity; showAll суммирует доходы. Пустой деструктор демонстрирует синтаксис; память string освобождается автоматически.

Проверки и ограничения. До 100 товаров; цена 0..1000000, количество 0..1000000. Максимальный модуль общей суммы не больше 10^14 и не переполняет double. Розничная цена ниже оптовой допустима: получается убыток, выводимый со знаком минус. double хранит приближённые десятичные значения; вывод округляется до двух знаков. Поиск цены сравнивает введённое значение, а не результат вычислений.

#### 3.2.1 Алгоритм

1. Создать массив объектов с помощью конструктора по умолчанию.

2. В меню выбрать добавление, вывод доходов, сортировку или поиск.

3. Проверить название, обе цены и количество.

4. Для каждого товара вычислить разницу цен, умноженную на количество.

5. Сложить доходы всех товаров и вывести таблицу.

6. Для сортировки сравнить выбранное поле; для поиска вывести все равные записи.

7. При выходе автоматически вызвать деструкторы объектов массива.

#### 3.2.2 Блок-схема

#### 3.2.3 Программный код

```cpp
#include "../common/input.h"

// Каждый объект хранит данные одного товара.
// Доход считаем простой формулой: разница цен * количество.
class Nomenclature {
  private:
    // Пользователь изменяет закрытые поля только через input.
    string name;
    double wholesale, retail;
    int quantity;

  public:
    // Конструктор создаёт объект с определёнными начальными значениями.
    Nomenclature() : name(""), wholesale(0), retail(0), quantity(0) {}
    Nomenclature(string n, double w, double r, int q)
        : name(n), wholesale(w), retail(r), quantity(q) {}
    // Динамическую память не выделяем, поэтому деструктор пустой.
    ~Nomenclature() {}

    void input() {
        // Отрицательные цены и количество отклоняет общий ввод.
        name = readText("Название товара: ", 80);
        wholesale = readDouble("Оптовая цена (0..1000000): ", 0, 1000000);
        retail = readDouble("Розничная цена (0..1000000): ", 0, 1000000);
        quantity = readInt("Количество (0..1000000): ", 0, 1000000);
    }

    double profit() const {
        double difference = retail - wholesale;
        return difference * quantity;
    }

    void show() const {
        // setprecision(2) оставляет две цифры после точки у цен и дохода.
        cout << fixed << setprecision(2) << name << " | " << wholesale << " | "
             << retail << " | " << quantity << " | " << profit() << '\n';
    }

    static void showAll(const Nomenclature goods[], int count) {
        // static-метод получает сразу весь массив товаров.
        cout << "Товар | Оптовая цена | Розничная цена | Количество | Доход\n";
        double total = 0;
        // Складываем доходы всех введённых товаров.
        for (int i = 0; i < count; i++) {
            goods[i].show();
            total += goods[i].profit();
        }
        if (count == 0) {
            cout << "Склад пуст.\n";
        }
        cout << fixed << setprecision(2) << "Общий доход: " << total << '\n';
    }

    double numberField(int field) const {
        // Эта функция выбирает числовое поле для поиска и сортировки.
        if (field == 2) return wholesale;
        if (field == 3) return retail;
        return quantity;
    }

    static void sort(Nomenclature goods[], int count, int field) {
        // Сравниваем соседние товары и меняем их местами.
        for (int i = 0; i < count - 1; i++) {
            for (int j = 0; j < count - 1 - i; j++) {
                bool exchange;
                if (field == 1) {
                    exchange = goods[j].name > goods[j + 1].name;
                } else {
                    exchange = goods[j].numberField(field) >
                               goods[j + 1].numberField(field);
                }
                if (exchange) {
                    // Временно сохраняем товар, чтобы поменять две записи
                    // местами.
                    Nomenclature temp = goods[j];
                    goods[j] = goods[j + 1];
                    goods[j + 1] = temp;
                }
            }
        }
        showAll(goods, count);
    }

    static void search(const Nomenclature goods[], int count, int field) {
        string text;
        double value = 0;
        if (field == 1) {
            text = readText("Название: ", 80);
        } else if (field == 4) {
            value = readInt("Количество: ", 0, 1000000);
        } else {
            value = readDouble("Цена: ", 0, 1000000);
        }
        bool found = false;
        // Сравниваем текст с текстом, а цену или количество — с числом.
        for (int i = 0; i < count; i++) {
            bool matches;
            if (field == 1) {
                matches = goods[i].name == text;
            } else {
                matches = goods[i].numberField(field) == value;
            }
            if (matches) {
                goods[i].show();
                found = true;
            }
        }
        if (!found) {
            cout << "Товар не найден.\n";
        }
    }
};

int main() {
    // При создании массива вызывается конструктор по умолчанию.
    Nomenclature goods[100];
    int count = 0;
    while (true) {
        cout << "\n1 - добавить товар, 2 - таблица и доход, 3 - сортировка, 4 "
                "- поиск, 0 - выход\n";
        int menu = readInt("Выбор: ", 0, 4);
        if (menu == 0) {
            break;
        }
        if (menu == 1) {
            if (count == 100) {
                cout << "Достигнут предел: 100 товаров.\n";
            } else {
                goods[count].input();
                // Новая запись занимает следующую свободную ячейку.
                count++;
            }
        }
        if (menu == 2) {
            Nomenclature::showAll(goods, count);
        }
        if (menu == 3 || menu == 4) {
            cout << "1 - название, 2 - оптовая цена, 3 - розничная цена, 4 - "
                    "количество\n";
            int field = readInt("Поле: ", 1, 4);
            if (menu == 3) {
                Nomenclature::sort(goods, count, field);
            } else {
                Nomenclature::search(goods, count, field);
            }
        }
    }
}
```

Условие задачи. Создать класс Soft с закрытыми названием программы, разработчиком, занимаемым объёмом и датой окончания лицензии. Использовать конструкторы и деструктор. Выводить сведения и число дней до завершения лицензии.

#include "../common/input.h"

// Номер дня с начала 1900 года позволяет вычитать даты без часовых поясов.
int dayNumber(const string &date) {
    // Вырезаем день, месяц и год из строки ДД.ММ.ГГГГ.
    int day = stoi(date.substr(0, 2));
    int month = stoi(date.substr(3, 2));
    int year = stoi(date.substr(6, 4));
    int result = day - 1;
    // Сначала прибавляем дни всех полных лет.
    for (int y = 1900; y < year; y++) {
        if (y % 400 == 0 || (y % 4 == 0 && y % 100 != 0)) {
            result += 366;
        } else {
            result += 365;
        }
    }
    int days[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    if (year % 400 == 0 || (year % 4 == 0 && year % 100 != 0)) {
        days[1] = 29;
    }
    // Затем прибавляем дни полных месяцев текущего года.
    for (int m = 1; m < month; m++) {
        result += days[m - 1];
    }
    return result;
}

class Soft {
  private:
    // В одном объекте находятся сведения об одной программе.
    string name, developer, endDate;
    double size;

  public:
    // Первый конструктор задаёт начальные значения, второй получает готовые
    // данные.
    Soft() : name(""), developer(""), endDate("01.01.1900"), size(0) {}
    Soft(string n, string d, double s, string date)
        : name(n), developer(d), endDate(date), size(s) {}
    ~Soft() {} // Поля string сами освобождают свою память.

    void input() {
        // Проверку числа дней в месяце выполняет readDate.
        name = readText("Название программы: ", 80);
        developer = readText("Разработчик: ", 80);
        size = readDouble("Объём в МБ (0..1000000000): ", 0, 1000000000);
        endDate = readDate("Дата окончания лицензии ДД.ММ.ГГГГ: ");
    }

    int daysLeft(const string &today) const {
        // Одинаковая дата даёт разность 0, прошедшая — отрицательную.
        int endDay = dayNumber(endDate);
        int currentDay = dayNumber(today);
        return endDay - currentDay;
    }

    void show(const string &today) const {
        int days = daysLeft(today);
        cout << name << " | " << developer << " | " << size << " МБ | "
             << endDate << " | ";
        // Знак разности показывает, истекла ли лицензия.
        if (days > 0) {
            cout << "осталось дней: " << days;
        } else if (days == 0) {
            cout << "истекает сегодня";
        } else {
            cout << "истекла дней назад: " << -days;
        }
        cout << '\n';
    }
};

                "выход\n";
        int menu = readInt("Выбор: ", 0, 2);
        if (menu == 0) {
            break;
        }
        if (menu == 1) {
            if (count == 100) {
                cout << "Достигнут предел: 100 программ.\n";
            } else {
                programs[count].input();
                count++;
            }
        }
        if (menu == 2) {
            if (count == 0) {
                cout << "Программ нет.\n";
                continue;
            }
            string today = readDate("Дата расчёта ДД.ММ.ГГГГ: ");
            // Все лицензии проверяем относительно одной даты.
            cout << "Программа | Разработчик | Объём | Лицензия | Срок\n";
            for (int i = 0; i < count; i++) {
                programs[i].show(today);
            }
        }
    }
}

            text.pop_back();
        }

        bool valid = true;
        if (text.empty()) {
            valid = false;
        }
        for (unsigned char c : text) {
            // Служебные символы не должны ломать таблицу в консоли.
            if (c < 32 || c == 127) {
                valid = false;
            }
        }
        if (valid) {
            return text;
        }
        cout << "Ошибка: введите непустой текст (до " << maxLength
             << " байт UTF-8).\n";
    }
}

inline int readInt(string prompt, int low, int high) {
    while (true) {
        string line = readLine(prompt);
        // Читаем число из всей строки, чтобы не принять 12abc за число 12.
        istringstream input(line);
        int value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        if (value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно целое число от " << low << " до " << high
             << ".\n";
    }
}

inline double readDouble(string prompt, double low, double high) {
    while (true) {
        istringstream input(readLine(prompt));
        double value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        // isfinite отсеивает бесконечность и недопустимый результат NaN.
        if (!isfinite(value) || value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно конечное число от " << low << " до " << high
             << ". Дробная часть отделяется точкой.\n";
    }
}

inline bool validDate(string date) {
    if (date.size() != 10 || date[2] != '.' || date[5] != '.') {
        return false;
    }
    for (int i = 0; i < 10; i++) {
        if (i != 2 && i != 5 && (date[i] < '0' || date[i] > '9')) {
            return false;
        }
    }
    // substr берёт часть строки, stoi переводит её в целое число.
    int day = stoi(date.substr(0, 2));
    int month = stoi(date.substr(3, 2));
    int year = stoi(date.substr(6, 4));
    if (year < 1900 || year > 2100 || month < 1 || month > 12) {
        return false;
    }
    int days[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    // Високосный год добавляет один день в феврале.
    if (year % 400 == 0 || (year % 4 == 0 && year % 100 != 0)) {
        days[1] = 29;
    }
    return day >= 1 && day <= days[month - 1];
}

// Запись ГГГГММДД позволяет правильно сравнить две даты как числа.
inline int dateNumber(string date) {
    return stoi(date.substr(6, 4)) * 10000 + stoi(date.substr(3, 2)) * 100 +
           stoi(date.substr(0, 2));
}

inline string readDate(string prompt) {
    while (true) {
        string date = readText(prompt, 10);
        if (validDate(date)) {
            return date;
        }
        cout << "Ошибка: нужна существующая дата ДД.ММ.ГГГГ, год 1900..2100.\n";
    }
}

inline string readTime(string prompt) {
    while (true) {
        string time = readText(prompt, 5);
        bool valid = time.size() == 5;
        if (valid) {
            valid = time[2] == ':';
            for (int i = 0; i < 5; i++) {
                if (i != 2 && (time[i] < '0' || time[i] > '9')) {
                    valid = false;
                }
            }
        }
        if (valid) {
            int hours = stoi(time.substr(0, 2));
            int minutes = stoi(time.substr(3, 2));
            if (hours <= 23 && minutes <= 59) {
                return time;
            }
        }
        cout << "Ошибка: время должно быть от 00:00 до 23:59.\n";
    }
}

inline string readPhone(string prompt) {
    while (true) {
        string phone = readText(prompt, 16);
        size_t start = 0;
        if (phone[0] == '+') {
            start = 1;
        }
        bool valid = phone.size() - start >= 7 && phone.size() - start <= 15;
        for (size_t i = start; i < phone.size(); i++) {
            if (phone[i] < '0' || phone[i] > '9') {
                valid = false;
            }
        }
        if (valid) {
            return phone;
        }
        cout << "Ошибка: нужно 7..15 цифр; знак + допустим только в начале.\n";
    }
}
