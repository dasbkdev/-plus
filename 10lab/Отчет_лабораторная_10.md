### 3.1 Безопасный массив

Условие задачи. Добавить к программе по мотивам ARROVER1 вложенный класс исключений, чтобы putel() и getel() генерировали исключение при выходе индекса за границы массива.

Данные программы. SafeArray: data[100], первоначально заполнен нулями. Вложенный Range хранит ошибочный index.

Проверки и ограничения. Разрешенные индексы: 0..99; для демонстрации исключения ввод допускает все int32. Значения: -2147483648..2147483647. Отклоненная запись не меняет массив.

#### 3.1.1 Алгоритм

1. Выбрать запись, чтение или просмотр крайних элементов.

2. Перед обращением к массиву проверить условие 0 <= index < 100.

3. При неверном индексе выполнить throw Range(index), иначе записать или прочитать элемент.

4. Перехватить Range, вывести индекс и допустимый диапазон, затем вернуться в меню.

Входные данные. Номер пункта меню и значения, которые запрашивает выбранная операция. SafeArray: data[100], первоначально заполнен нулями. Вложенный Range хранит ошибочный index.

Выходные данные. Результат выбранного действия, текущее состояние данных или точное сообщение об ошибке. Проверить индекс 0..99; при нарушении вызвать Range, иначе обратиться к массиву.

#### 3.1.2 Блок-схема

#### 3.1.3 Программный код

```cpp
#include "../common/input.h"

// Пользователь записывает и читает элементы массива из 100 чисел.
// Неверный индекс вызывает исключение до обращения к памяти.
// Обработчик показывает ошибку, после чего меню работает дальше.
class SafeArray {
  private:
    int data[100] = {};

  public:
    class Range {
        // Объект ошибки запоминает индекс, который ввел пользователь.
      public:
        int index;
        explicit Range(int i) : index(i) {}
    };
    void putel(int index, int value) {
        // Проверка должна идти перед записью в массив.
        if (index < 0 || index >= 100) throw Range(index);
        data[index] = value;
    }
    int getel(int index) const {
        if (index < 0 || index >= 100) throw Range(index);
        return data[index];
    }
};

int main() {
    SafeArray array;
    while (true) {
        cout << "\n1. Записать элемент\n2. Прочитать элемент\n3. Показать края "
                "массива\n0. Выход\n";
        int menu = readInt("Выбор: ", 0, 3);
        if (menu == 0) break;
        // В try находятся действия, которые могут выбросить Range.
        try {
            if (menu == 1 || menu == 2) {
                int index = readInt(
                    "Индекс (допустимы 0..99): ", -2147483647 - 1, 2147483647);
                if (menu == 1) {
                    int value =
                        readInt("Значение: ", -2147483647 - 1, 2147483647);
                    array.putel(index, value);
                    cout << "Элемент записан.\n";
                } else {
                    int value = array.getel(index);
                    cout << "Элемент: " << value << '\n';
                }
            }
            if (menu == 3) {
                cout << "a[0] = " << array.getel(0);
                cout << "; a[99] = " << array.getel(99) << '\n';
            }
        } catch (const SafeArray::Range &error) {
            // const-ссылка дает прочитать ошибку без создания ее копии.
            // После обработки ошибки меню продолжает работу.
            cout << "Исключение Range: индекс " << error.index
                 << " вне диапазона 0..99.\n";
        }
    }
}
```

### 3.2 Площадь прямоугольника с исключениями

Условие задачи. Выполнить задачу 3 лабораторной 5 с механизмом исключений: перегрузить rect_area(a,b) для прямоугольника и rect_area(side) для квадрата, отклонять недопустимые размеры и результат.

Данные программы. Rectangle не хранит состояние: результат определяется аргументами. Вложенный SizeError обозначает неверные размеры или непредставимую положительную площадь.

Проверки и ограничения. Ввод сторон: -1000000000..1000000000; неположительные значения допускаются на этапе ввода для демонстрации исключения. NaN и бесконечность отклоняет общий ввод; слишком малое произведение, обнулившееся в double, вызывает SizeError.

#### 3.2.1 Алгоритм

1. Выбрать прямоугольник или квадрат и прочитать аргументы.

2. Проверить конечность и положительность сторон, конечность и положительность их произведения.

3. Вызвать подходящую перегрузку rect_area; квадрат использует общую реализацию с двумя одинаковыми сторонами.

4. Перехватить SizeError и продолжить меню без ложного результата.

Входные данные. Номер пункта меню и значения, которые запрашивает выбранная операция. Rectangle не хранит состояние: результат определяется аргументами. Вложенный SizeError обозначает неверные размеры или непредставимую положительную площадь.

Выходные данные. Результат выбранного действия, текущее состояние данных или точное сообщение об ошибке. Проверить положительную конечную площадь; вернуть a*b либо вызвать SizeError.

#### 3.2.2 Блок-схема

#### 3.2.3 Программный код

```cpp
#include "../common/input.h"
#include <cmath>

// Один метод считает площадь прямоугольника, другой — квадрата.
// Одинаковое имя с разными аргументами называется перегрузкой.
// Неправильные стороны обрабатываются через исключение.
class Rectangle {
  public:
    class SizeError {};
    double rect_area(double a, double b) const {
        if (!isfinite(a) || !isfinite(b)) throw SizeError();
        if (a <= 0 || b <= 0) throw SizeError();
        double area = a * b;
        // Слишком большое или слишком малое произведение нельзя выдавать за
        // площадь.
        if (!isfinite(area) || area <= 0) throw SizeError();
        return area;
    }
    double rect_area(double side) const {
        return rect_area(
            side, side); // Перегрузка для квадрата использует общую проверку.
    }
};

int main() {
    Rectangle figure;
    while (true) {
        cout << "\n1. Площадь прямоугольника\n2. Площадь квадрата\n0. Выход\n";
        int menu = readInt("Выбор: ", 0, 2);
        if (menu == 0) break;
        double a = readDouble(
            "Первая сторона (должна быть положительной): ", -1e9, 1e9);
        try {
            // Вызывается перегрузка с одним или с двумя размерами.
            double result;
            if (menu == 1) {
                double b = readDouble("Вторая сторона: ", -1e9, 1e9);
                result = figure.rect_area(a, b);
            } else result = figure.rect_area(a);
            cout << "Площадь: " << result << '\n';
        } catch (const Rectangle::SizeError &) {
            // После catch цикл снова покажет меню.
            cout << "Исключение SizeError: стороны и площадь должны быть "
                    "положительными и конечными.\n";
        }
    }
}
```

### 3.3 Исключения целочисленного стека

Условие задачи. Добавленное корректировкой задание темы «Исключения»: создать Stack емкостью 3 с массивом st[max], индексом top и вложенным Rang. При push в полный стек и pop из пустого генерировать исключение.

Данные программы. Stack: static const int max = 3, st[max], top = -1. Stack::Rang содержит текст причины ошибки.

Проверки и ограничения. Ровно 3 элемента; целочисленные значения в диапазоне int32. Повторное удаление из пустого стека и добавление четвертого элемента не меняют данные.

#### 3.3.1 Алгоритм

1. Показать меню push, pop и просмотр стека.

2. Перед push проверить свободное место; перед pop проверить наличие элемента.

3. Генерировать Rang до изменения top при нарушении границ.

4. Перехватить исключение, вывести причину и сохранить предыдущее состояние.

Входные данные. Номер пункта меню и значения, которые запрашивает выбранная операция. Stack: static const int max = 3, st[max], top = -1. Stack::Rang содержит текст причины ошибки.

Выходные данные. Результат выбранного действия, текущее состояние данных или точное сообщение об ошибке. Проверить границы стека из трех элементов; выполнить push/pop либо вызвать Rang.

#### 3.3.2 Блок-схема

#### 3.3.3 Программный код

```cpp
#include "../common/input.h"

// Стек помещает не больше трех целых чисел.
// top хранит индекс вершины; -1 означает пустой стек.
// Исключение не дает добавить четвертое число или удалить отсутствующее.
class Stack {
  private:
    static const int max = 3;
    int st[max] = {};
    int top = -1;

  public:
    class Rang {
        // Вложенный класс описывает ошибку именно этого стека.
      public:
        string message;
        explicit Rang(string text) : message(text) {}
    };
    void push(int value) {
        if (top == max - 1) throw Rang("стек полон");
        top++;
        st[top] = value;
    }
    int pop() {
        if (top == -1) throw Rang("стек пуст");
        int value = st[top];
        top--; // Вершина меняется только после успешной проверки.
        return value;
    }
    void show() const {
        cout << "Стек от вершины: ";
        if (top == -1) cout << "пусто";
        // Идем от вершины к первому добавленному элементу.
        for (int i = top; i >= 0; i--) cout << st[i] << ' ';
        cout << '\n';
    }
};

int main() {
    Stack stack;
    while (true) {
        cout << "\n1. Push\n2. Pop\n3. Показать стек\n0. Выход\n";
        int menu = readInt("Выбор: ", 0, 3);
        if (menu == 0) break;
        try {
            if (menu == 1) {
                int value =
                    readInt("Целое число: ", -2147483647 - 1, 2147483647);
                stack.push(value);
                cout << "Элемент добавлен.\n";
            }
            if (menu == 2) {
                int value = stack.pop();
                cout << "Удален: " << value << '\n';
            }
            if (menu == 3) stack.show();
        } catch (const Stack::Rang &error) {
            // throw передает сюда причину ошибки; const& не копирует объект.
            cout << "Исключение Rang: " << error.message
                 << ". Состояние стека сохранено.\n";
        }
    }
}
```

inline string readText(string prompt, size_t maxLength = 100) {
    while (true) {
        string text = readLine(prompt);
        if (text.size() > maxLength) {
            cout << "Ошибка: введите непустой текст (до " << maxLength
                 << " байт UTF-8).\n";
            continue;
        }
        // Убираем пробелы в начале и конце ответа.
        while (!text.empty() && (text.front() == ' ' || text.front() == '\t' ||
                                 text.front() == '\r')) {
            text.erase(0, 1);
        }
        while (!text.empty() && (text.back() == ' ' || text.back() == '\t' ||
                                 text.back() == '\r')) {
            text.pop_back();
        }

        bool valid = true;
        if (text.empty()) {
            valid = false;
        }
        for (unsigned char c : text) {
            // Служебные символы не должны ломать таблицу в консоли.
            if (c < 32 || c == 127) {
                valid = false;
            }
        }
        if (valid) {
            return text;
        }
        cout << "Ошибка: введите непустой текст (до " << maxLength
             << " байт UTF-8).\n";
    }
}

inline int readInt(string prompt, int low, int high) {
    while (true) {
        string line = readLine(prompt);
        // Читаем число из всей строки, чтобы не принять 12abc за число 12.
        istringstream input(line);
        int value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        if (value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно целое число от " << low << " до " << high
             << ".\n";
    }
}

inline double readDouble(string prompt, double low, double high) {
    while (true) {
        istringstream input(readLine(prompt));
        double value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        // isfinite отсеивает бесконечность и недопустимый результат NaN.
        if (!isfinite(value) || value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно конечное число от " << low << " до " << high
             << ". Дробная часть отделяется точкой.\n";
    }
}

inline bool validDate(string date) {
    if (date.size() != 10 || date[2] != '.' || date[5] != '.') {
        return false;
    }
    for (int i = 0; i < 10; i++) {
        if (i != 2 && i != 5 && (date[i] < '0' || date[i] > '9')) {
            return false;
        }
    }
    // substr берёт часть строки, stoi переводит её в целое число.
    int day = stoi(date.substr(0, 2));
    int month = stoi(date.substr(3, 2));
    int year = stoi(date.substr(6, 4));
    if (year < 1900 || year > 2100 || month < 1 || month > 12) {
        return false;
    }
    int days[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    // Високосный год добавляет один день в феврале.
    if (year % 400 == 0 || (year % 4 == 0 && year % 100 != 0)) {
        days[1] = 29;
    }
    return day >= 1 && day <= days[month - 1];
}

           stoi(date.substr(0, 2));
}

        string time = readText(prompt, 5);

        if (valid) {
