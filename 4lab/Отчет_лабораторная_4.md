### 3.1 Обычная функция сравнения скоростей

Условие задачи. Создать классы Kamaz и Tatra с закрытыми полями марки и скоростей. Обычная функция FrCreater сравнивает скорости на трёх участках через открытые методы чтения. Вывести таблицу и определить победителя.

Данные программы. Kamaz и Tatra: name — закрытая строка марки; speed[3] — закрытые скорости на равнине, в горах и пустыне. В main: два объекта грузовиков, distance[3] — длины общих участков; ready — признак заполненных данных.

Проверки и ограничения. Длина каждого участка: 0.1..100000 км. Скорость: 0.1..300 км/ч. Границы положительны, поэтому деление на ноль исключено. Название: непустое, до 60 байт UTF-8. Числа должны быть конечными, без постороннего хвоста. Неверная строка запрашивается повторно. Маршрут одинаков для обеих машин. Время вычисляется как сумма distance[i]/speed[i]. Скорости не суммируются для определения победителя. Разница суммарного времени не более 0.000000001 часа считается ничьей. До ввода всех данных таблица недоступна. EOF завершает программу.

#### 3.1.1 Алгоритм

1. Повторять меню до команды 0; команда 1 вводит длины трёх участков, марки машин и скорости на участках.

2. Проверить диапазоны и завершённость ввода; сохранить данные в закрытых полях через методы.

3. Для каждого участка вызвать FrCreater и вывести строку таблицы: 1 при большей скорости КАМАЗа, 0 при равенстве, -1 при большей скорости Tatra.

4. Для каждой машины сложить три времени движения, сравнить итоговые времена и вывести победителя или ничью.

5. Обычная FrCreater читает скорости через getSpeed, поэтому прямого доступа к закрытым полям у неё нет.

#include "../common/input.h"

const string stages[3] = {"Равнина", "Горы", "Пустыня"};
class Tatra;

class Kamaz {
    string name; // Марка хранится в закрытом поле.
    double speed[3]; // Скорости на трёх участках маршрута.

  public:
    Kamaz() : name("КАМАЗ"), speed{0, 0, 0} {}
    // Вводим только положительную скорость, чтобы время можно было вычислить.
    void set() {
        name = readText("Марка КАМАЗа: ", 60);
        for (int i = 0; i < 3; i++) {
            speed[i] = readDouble(
                stages[i] + ", скорость КАМАЗ (0.1..300 км/ч): ", 0.1, 300);
        }
    }
    // const означает, что метод не меняет данные машины.
    string getName() const { return name; }
    double getSpeed(int i) const { return speed[i]; }
    void show(int i) const { cout << speed[i]; }
    double time(const double distance[]) const {
        double hours = 0;
        // Время всего маршрута равно сумме времён трёх участков.
        for (int i = 0; i < 3; i++) hours += distance[i] / speed[i];
        return hours;
    }
};

class Tatra {
    string name; // Марка хранится в закрытом поле.
    double speed[3]; // Скорости на трёх участках маршрута.

  public:
    Tatra() : name("Tatra"), speed{0, 0, 0} {}
    // Вводим только положительную скорость, чтобы время можно было вычислить.
    void set() {
        name = readText("Марка Tatra: ", 60);
        for (int i = 0; i < 3; i++) {
            speed[i] = readDouble(
                stages[i] + ", скорость Tatra (0.1..300 км/ч): ", 0.1, 300);
        }
    }
    // const означает, что метод не меняет данные машины.
    string getName() const { return name; }
    double getSpeed(int i) const { return speed[i]; }
    void show(int i) const { cout << speed[i]; }
    double time(const double distance[]) const {
        double hours = 0;
        for (int i = 0; i < 3; i++) hours += distance[i] / speed[i];
        return hours;
    }
};

// const-ссылка передаёт исходный объект без копирования и изменения.
int FrCreater(const Kamaz &kamaz, const Tatra &tatra, int i) {
    if (kamaz.getSpeed(i) > tatra.getSpeed(i)) return 1;
    if (kamaz.getSpeed(i) < tatra.getSpeed(i)) return -1;
    return 0;
}

int main() {
    Kamaz kamaz;
    Tatra tatra;
    double distance[3] = {};
    bool ready = false; // Пока данные не введены, результат считать нельзя.

    cout << "Обычная функция FrCreater получает скорости через getters.\n";
    cout << "Сравнение: 1 — быстрее КАМАЗ, 0 — равны, -1 — быстрее Tatra.\n";
    while (true) {
        cout << "\n1 Ввести данные  2 Таблица и победитель  0 Выход\n";
        int choice = readInt("Команда: ", 0, 2);
        if (choice == 0) break;

        if (choice == 1) {
            cout << "Обе машины проходят одни и те же три участка.\n";
            // В массиве по одной длине для равнины, гор и пустыни.
            for (int i = 0; i < 3; i++) {
                distance[i] = readDouble(
                    stages[i] + ", длина (0.1..100000 км): ", 0.1, 100000);
            }
            kamaz.set();
            tatra.set();
            ready = true;
        }

        if (choice == 2) {
            if (!ready) {
                cout << "Сначала введите данные командой 1.\n";
                continue; // Возвращаемся к меню без вычислений.
            }
            cout << fixed << setprecision(3);
            cout << "\nКАМАЗ: " << kamaz.getName()
                 << "; Tatra: " << tatra.getName() << '\n';
            cout << "Участок | км | КАМАЗ км/ч | Tatra км/ч | FrCreater\n";
            // Каждая итерация печатает результаты одного участка.
            for (int i = 0; i < 3; i++) {
                cout << stages[i] << " | " << distance[i] << " | ";
                kamaz.show(i);
                cout << " | ";
                tatra.show(i);
                cout << " | " << FrCreater(kamaz, tatra, i) << '\n';
            }
            double kamazHours = kamaz.time(distance);
            double tatraHours = tatra.time(distance);
            cout << setprecision(9) << "Время КАМАЗ (ч): " << kamazHours
                 << "\nВремя Tatra (ч): " << tatraHours << '\n';

            // Из-за округления double очень близкие времена считаем равными.
            if (fabs(kamazHours - tatraHours) <= 1e-9) {
                cout << "Итог: ничья (разница времени не больше 0.000000001 "
                        "ч).\n";
            } else if (kamazHours < tatraHours) {
                cout << "Победитель: КАМАЗ (" << kamaz.getName() << ")\n";
            } else {
                cout << "Победитель: Tatra (" << tatra.getName() << ")\n";
            }
        }
    }
    return 0;
}

### 3.3 Дружественный класс Tatra

Условие задачи. Объявить Tatra дружественным классу Kamaz. Сравнивать скорости методом Tatra::FrCreater, вывести таблицу и победителя общего маршрута.

Данные программы. Kamaz и Tatra: name — закрытая строка марки; speed[3] — закрытые скорости на равнине, в горах и пустыне. В main: два объекта грузовиков, distance[3] — длины общих участков; ready — признак заполненных данных.

Проверки и ограничения. Длина каждого участка: 0.1..100000 км. Скорость: 0.1..300 км/ч. Границы положительны, поэтому деление на ноль исключено. Название: непустое, до 60 байт UTF-8. Числа должны быть конечными, без постороннего хвоста. Неверная строка запрашивается повторно. Маршрут одинаков для обеих машин. Время вычисляется как сумма distance[i]/speed[i]. Скорости не суммируются для определения победителя. Разница суммарного времени не более 0.000000001 часа считается ничьей. До ввода всех данных таблица недоступна. EOF завершает программу.

#### 3.3.1 Алгоритм

1. Повторять меню до команды 0; команда 1 вводит длины трёх участков, марки машин и скорости на участках.

2. Проверить диапазоны и завершённость ввода; сохранить данные в закрытых полях через методы.

3. Для каждого участка вызвать FrCreater и вывести строку таблицы: 1 при большей скорости КАМАЗа, 0 при равенстве, -1 при большей скорости Tatra.

4. Для каждой машины сложить три времени движения, сравнить итоговые времена и вывести победителя или ничью.

5. В Kamaz задано friend class Tatra. Метод Tatra::FrCreater читает закрытую скорость переданного Kamaz и скорость своего объекта. Дружба действует только в указанном направлении.

#include "../common/input.h"

const string stages[3] = {"Равнина", "Горы", "Пустыня"};
class Tatra;

class Kamaz {
    string name; // Марка хранится в закрытом поле.
    double speed[3]; // Скорости на трёх участках маршрута.
    friend class Tatra; // Методы Tatra получают доступ к закрытым полям Kamaz.
  public:
    Kamaz() : name("КАМАЗ"), speed{0, 0, 0} {}
    // Вводим только положительную скорость, чтобы время можно было вычислить.
    void set() {
        name = readText("Марка КАМАЗа: ", 60);
        for (int i = 0; i < 3; i++) {
            speed[i] = readDouble(
                stages[i] + ", скорость КАМАЗ (0.1..300 км/ч): ", 0.1, 300);
        }
    }
    // const означает, что метод не меняет данные машины.
    string getName() const { return name; }
    double getSpeed(int i) const { return speed[i]; }
    void show(int i) const { cout << speed[i]; }
    double time(const double distance[]) const {
        double hours = 0;
        // Время всего маршрута равно сумме времён трёх участков.
        for (int i = 0; i < 3; i++) hours += distance[i] / speed[i];
        return hours;
    }
};

class Tatra {
    string name; // Марка хранится в закрытом поле.
    double speed[3]; // Скорости на трёх участках маршрута.

  public:
    Tatra() : name("Tatra"), speed{0, 0, 0} {}
    // Вводим только положительную скорость, чтобы время можно было вычислить.
    void set() {
        name = readText("Марка Tatra: ", 60);
        for (int i = 0; i < 3; i++) {
            speed[i] = readDouble(
                stages[i] + ", скорость Tatra (0.1..300 км/ч): ", 0.1, 300);
        }
    }
    // const означает, что метод не меняет данные машины.
    string getName() const { return name; }
    double getSpeed(int i) const { return speed[i]; }
    void show(int i) const { cout << speed[i]; }
    double time(const double distance[]) const {
        double hours = 0;
        for (int i = 0; i < 3; i++) hours += distance[i] / speed[i];
        return hours;
    }
    // const-ссылка передаёт исходный объект без копирования и изменения.
    int FrCreater(const Kamaz &kamaz, int i) const {
        if (kamaz.speed[i] > speed[i]) return 1;
        if (kamaz.speed[i] < speed[i]) return -1;
        return 0;
    }
};

int main() {
    Kamaz kamaz;
    Tatra tatra;
    double distance[3] = {};
    bool ready = false; // Пока данные не введены, результат считать нельзя.

    cout << "Tatra — дружественный класс для Kamaz; сравнение выполняет метод "
            "Tatra.\n";
    cout << "Сравнение: 1 — быстрее КАМАЗ, 0 — равны, -1 — быстрее Tatra.\n";
    while (true) {
        cout << "\n1 Ввести данные  2 Таблица и победитель  0 Выход\n";
        int choice = readInt("Команда: ", 0, 2);
        if (choice == 0) break;

        if (choice == 1) {
            cout << "Обе машины проходят одни и те же три участка.\n";
            // В массиве по одной длине для равнины, гор и пустыни.
            for (int i = 0; i < 3; i++) {
                distance[i] = readDouble(
                    stages[i] + ", длина (0.1..100000 км): ", 0.1, 100000);
            }
            kamaz.set();
            tatra.set();
            ready = true;
        }

        if (choice == 2) {
            if (!ready) {
                cout << "Сначала введите данные командой 1.\n";
                continue; // Возвращаемся к меню без вычислений.
            }
            cout << fixed << setprecision(3);
            cout << "\nКАМАЗ: " << kamaz.getName()
                 << "; Tatra: " << tatra.getName() << '\n';
            cout << "Участок | км | КАМАЗ км/ч | Tatra км/ч | "
                    "Tatra::FrCreater\n";
            // Каждая итерация печатает результаты одного участка.
            for (int i = 0; i < 3; i++) {
                cout << stages[i] << " | " << distance[i] << " | ";
                kamaz.show(i);
                cout << " | ";
                tatra.show(i);
                cout << " | " << tatra.FrCreater(kamaz, i) << '\n';
            }
            double kamazHours = kamaz.time(distance);
            double tatraHours = tatra.time(distance);
            cout << setprecision(9) << "Время КАМАЗ (ч): " << kamazHours
                 << "\nВремя Tatra (ч): " << tatraHours << '\n';

            // Из-за округления double очень близкие времена считаем равными.
            if (fabs(kamazHours - tatraHours) <= 1e-9) {
                cout << "Итог: ничья (разница времени не больше 0.000000001 "
                        "ч).\n";
            } else if (kamazHours < tatraHours) {
                cout << "Победитель: КАМАЗ (" << kamaz.getName() << ")\n";
            } else {
                cout << "Победитель: Tatra (" << tatra.getName() << ")\n";
            }
        }
    }
    return 0;

inline string readText(string prompt, size_t maxLength = 100) {
    while (true) {
        string text = readLine(prompt);
        if (text.size() > maxLength) {
            cout << "Ошибка: введите непустой текст (до " << maxLength
                 << " байт UTF-8).\n";
            continue;
        }
        // Убираем пробелы в начале и конце ответа.
        while (!text.empty() && (text.front() == ' ' || text.front() == '\t' ||
                                 text.front() == '\r')) {
            text.erase(0, 1);
        }
        while (!text.empty() && (text.back() == ' ' || text.back() == '\t' ||
                                 text.back() == '\r')) {
            text.pop_back();
        }

        bool valid = true;
        if (text.empty()) {
            valid = false;
        }
        for (unsigned char c : text) {
            // Служебные символы не должны ломать таблицу в консоли.
            if (c < 32 || c == 127) {
                valid = false;
            }
        }
        if (valid) {
            return text;
        }
        cout << "Ошибка: введите непустой текст (до " << maxLength
             << " байт UTF-8).\n";
    }
}

inline int readInt(string prompt, int low, int high) {
    while (true) {
        string line = readLine(prompt);
        // Читаем число из всей строки, чтобы не принять 12abc за число 12.
        istringstream input(line);
        int value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        if (value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно целое число от " << low << " до " << high
             << ".\n";
    }
}

inline double readDouble(string prompt, double low, double high) {
    while (true) {
        istringstream input(readLine(prompt));
        double value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        // isfinite отсеивает бесконечность и недопустимый результат NaN.

            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно конечное число от " << low << " до " << high
             << ". Дробная часть отделяется точкой.\n";
    }
}

inline bool validDate(string date) {
    if (date.size() != 10 || date[2] != '.' || date[5] != '.') {
        return false;
    }
    for (int i = 0; i < 10; i++) {
        if (i != 2 && i != 5 && (date[i] < '0' || date[i] > '9')) {
            return false;
        }
    }
    // substr берёт часть строки, stoi переводит её в целое число.
    int day = stoi(date.substr(0, 2));
    int month = stoi(date.substr(3, 2));
    int year = stoi(date.substr(6, 4));
    if (year < 1900 || year > 2100 || month < 1 || month > 12) {
        return false;
    }
    int days[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    // Високосный год добавляет один день в феврале.
    if (year % 400 == 0 || (year % 4 == 0 && year % 100 != 0)) {
        days[1] = 29;
    }
    return day >= 1 && day <= days[month - 1];
}
