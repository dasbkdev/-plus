## 3 ХОД РАБОТЫ

### 3.1 Система «Платежи»

Условие задачи. Клиент связан со счетом и банковской картой. Реализовать оплату заказа, перевод другому клиенту, блокировку карты, аннулирование счета и блокировку администратором за превышение остатка. Определить вложенные классы и связи композиции и агрегации.

Данные программы. Account хранит номер, баланс в копейках и признак активности. Client: имя, указатель на счет и вложенная карта Card с признаком блокировки. Administrator блокирует карту при превышении остатка.

Проверки и ограничения. Операция: 0.01..1000000000.00; остаток не выше 1000000000.00. Ввод: до двух знаков после точки, без экспоненты, минуса, NaN и посторонних символов. Два учебных клиента; перевод только другому. Закрытый счет не восстанавливается, карта не разблокируется.

#### 3.1.1 Алгоритм

1. Создать счета 1001 и 1002 с остатками 1000.00 и 500.00; связать их с клиентами.

2. Выбрать клиента и операцию; перевести сумму в целые копейки.

3. Для платежа проверить счет, карту, остаток и счет получателя. Изменить оба остатка только после проверок; при ошибке деньги сохранить.

4. При превышении остатка заблокировать карту. Закрывать только открытый счет с нулевым остатком, одновременно блокируя карту.

5. Пополнение и кассовое снятие выполнять без карты, только по открытому счету. Так можно обнулить счет и после блокировки карты.

Входные данные. Пункт меню, номер клиента, название заказа и сумма операции.

Выходные данные. Состояние счетов и карт, результат операции или сообщение об ошибке.

```cpp
#include "../common/input.h"
#include <iomanip>

// Два клиента могут оплачивать заказы и переводить деньги друг другу.
// Счет существует отдельно, а карта хранится внутри клиента.
// Все суммы считаются в целых копейках, чтобы не терять дробные деньги.
const long long MAX_MONEY = 100000000000LL;

// Разбираем сумму как текст: например, 12.34 превращается в 1234 копейки.
long long readMoney(const string &prompt) {
    while (true) {
        string text = readText(prompt, 100);
        bool valid = true;
        long long cents = 0;
        int decimals = 0;
        bool point = false;
        for (char symbol : text) {
            if (symbol == '.' && !point) {
                point = true;
                continue;
            }
            if (symbol < '0' || symbol > '9') {
                valid = false;
                break;
            }
            // Приписываем очередную цифру справа к уже собранному числу.
            cents = cents * 10 + symbol - '0';
            if (point) decimals++;
            if (cents > MAX_MONEY || decimals > 2) {
                valid = false;
                break;
            }
        }
        if (text.empty() || text.front() == '.' || text.back() == '.')
            valid = false;
        if (valid) {
            // Для 12 добавляем два нуля, для 12.3 — один ноль.
            for (int i = decimals; i < 2; i++) cents *= 10;
            if (cents > 0 && cents <= MAX_MONEY) return cents;
        }
        cout << "Введите сумму 0.01..1000000000, не более двух знаков после "
                "точки.\n";
    }
}

void showMoney(long long cents) {
    // Деление дает целую часть, остаток от деления — копейки.
    cout << cents / 100 << '.' << setw(2) << setfill('0') << cents % 100
         << setfill(' ');
}

class Account {
  private:
    int number;
    // Целое число здесь означает именно копейки, а не целые денежные единицы.
    long long balance;
    bool active;

  public:
    Account(int n, long long b) : number(n), balance(b), active(true) {}
    bool isActive() const { return active; }
    long long getBalance() const { return balance; }
    void show() const {
        cout << "Счет " << number;
        if (active) cout << " открыт; остаток: ";
        else cout << " закрыт; остаток: ";
        showMoney(balance);
        cout << '\n';
    }
    bool deposit(long long amount) {
        if (!active) return false;
        if (amount > MAX_MONEY - balance) return false;
        balance += amount;
        return true;
    }
    bool withdraw(long long amount) {
        if (!active) return false;
        if (amount > balance) return false;
        balance -= amount;
        return true;
    }
    bool transfer(Account &other, long long amount) {
        // other — ссылка на счет получателя, this — адрес текущего счета.
        if (this == &other) return false;
        if (!active || !other.active) return false;
        if (amount > balance) return false;
        if (amount > MAX_MONEY - other.balance) return false;
        balance -= amount; // Проверки завершены до изменения двух счетов.
        other.balance += amount;
        return true;
    }
    bool cancel() {
        // Закрывать счет с оставшимися деньгами нельзя.
        if (!active) return false;
        if (balance != 0) return false;
        active = false;
        return true;
    }
};

class Client {
  private:
    class Card {
      private:
        bool blocked = false;

      public:
        bool isBlocked() const { return blocked; }
        void block() { blocked = true; }
    };
    string name;
    Account *account; // Агрегация: счет существует независимо от клиента.
    Card card; // Композиция: карта является частью клиента.
  public:
    Client(string n, Account &a) : name(n), account(&a) {}
    void show() const {
        cout << "Клиент: " << name << "; карта: ";
        if (card.isBlocked()) cout << "заблокирована\n";
        else cout << "активна\n";
        account->show();
    }
    Account &getAccount() { return *account; }
    bool canPay() const {
        // Стрелка -> вызывает метод объекта по сохраненному указателю.
        if (!account->isActive()) {
            cout << "Счет закрыт.\n";
            return false;
        }
        if (card.isBlocked()) {
            cout << "Карта заблокирована.\n";
            return false;
        }
        return true;
    }
    void blockCard() { card.block(); }
    void cancelAccount() {
        if (account->cancel()) {
            card.block();
            cout << "Счет аннулирован, карта заблокирована.\n";
        } else
            cout << "Нельзя закрыть: счет уже закрыт или его остаток не равен "
                    "нулю.\n";
    }
};

class Administrator {
  public:
    void blockForExcess(Client &client) const {
        // Блокировка запрещает оплату картой, но не уничтожает остаток счета.
        client.blockCard();
        cout << "Платеж превышает остаток. Администратор заблокировал карту. "
                "Деньги не списаны.\n";
    }
};

int main() {
    Account first(1001, 100000), second(1002, 50000);
    Client clients[2] = {Client("Клиент 1", first), Client("Клиент 2", second)};
    Administrator admin;
    int current = 0;
    cout << "Учебные счета: 1001 = 1000.00; 1002 = 500.00.\n";
    while (true) {
        cout << "\nТекущий клиент: " << current + 1
             << "\n1. Все счета\n2. Пополнить счет\n"
             << "3. Оплатить заказ\n4. Перевести другому клиенту\n5. "
                "Заблокировать карту\n"
             << "6. Аннулировать счет\n7. Сменить клиента\n8. Снять в кассе "
                "банка\n0. Выход\n";
        int menu = readInt("Выбор: ", 0, 8);
        if (menu == 0) break;
        // Ссылки позволяют работать с выбранным клиентом без его копии.
        Client &client = clients[current];
        Account &account = client.getAccount();
        if (menu == 1) {
            clients[0].show();
            clients[1].show();
        }
        if (menu == 7) current = readInt("Номер клиента (1 или 2): ", 1, 2) - 1;
        if (menu == 5) {
            client.blockCard();
            cout << "Карта заблокирована клиентом.\n";
        }
        if (menu == 6) client.cancelAccount();
        if (menu == 2 || menu == 8) {
            if (!account.isActive()) {
                cout << "Счет закрыт.\n";
                continue;
            }
            long long amount = readMoney("Сумма: ");
            bool ok;
            if (menu == 2) ok = account.deposit(amount);
            else ok = account.withdraw(amount);
            if (ok) cout << "Операция выполнена.\n";
            else cout << "Недостаточно средств или превышен предел остатка.\n";
        }
        if (menu == 3 || menu == 4) {
            if (!client.canPay()) continue;
            if (menu == 3) readText("Название заказа: ");
            long long amount = readMoney("Сумма платежа: ");
            if (amount > account.getBalance()) {
                admin.blockForExcess(client);
                continue;
            }
            bool ok;
            if (menu == 3) ok = account.withdraw(amount);
            else {
                int receiver = 1 - current;
                ok = account.transfer(clients[receiver].getAccount(), amount);
            }
            if (ok) cout << "Платеж выполнен.\n";
            else
                cout << "Перевод отклонен: счет получателя закрыт или "
                        "достигнут предел остатка.\n";
        }
    }
}
```

### 3.2 Система «Автобаза»

Условие задачи. Диспетчер назначает водителя и автомобиль на рейс, может отстранять водителя. Водитель подает заявку на ремонт и отмечает выполнение рейса и состояние автомобиля. Реализовать меню, закрытые поля, вложенный класс и согласованные состояния.

Данные программы. Driver: name, suspended, busy. Car: number, repair, busy. Depot содержит указатели на независимые массивы водителей и автомобилей, количества записей и собственный массив вложенных Trip. Trip: route, driver, car, done.

Проверки и ограничения. До 100 водителей, 100 автомобилей и 100 рейсов за запуск. Номера записей выбираются только из существующих; удаление записей не предусмотрено условием. ФИО/маршрут до 100 байт UTF-8, номер автомобиля до 20 байт UTF-8. Учет хранится в памяти до завершения программы. Физические документы о рейсах и ремонте не моделируются.

#### 3.2.1 Алгоритм

1. Добавить водителей и автомобили в пустые списки.

2. Перед назначением проверить вместимость, допуск и занятость водителя, исправность и занятость автомобиля.

3. Создать рейс и одновременно пометить водителя и автомобиль занятыми.

4. При завершении освободить водителя и автомобиль, сохранить отметку об исправности.

5. Отстранение запрещает новые рейсы; текущий рейс разрешено завершить.

6. Заявка на ремонт блокирует свободный автомобиль для назначения. Для машины в рейсе состояние ремонта отмечается при завершении этого рейса.

7. После ремонта диспетчер отмечает автомобиль исправным. Повторное завершение рейса запрещено.

Входные данные. Пункт меню; имена водителей, номера автомобилей, маршруты и номера выбранных записей.

Выходные данные. Списки водителей, автомобилей и рейсов; текущие состояния и сообщения об ошибках.

```cpp
#include "../common/input.h"

// Автобаза хранит водителей, автомобили и назначенные рейсы.
// Перед назначением проверяем, свободны ли водитель и машина.
// Выполненный рейс освобождает их для следующего задания.
class Driver {
  private:
    string name;
    // Отстранение и занятость — разные причины запрета нового рейса.
    bool suspended = false;
    bool busy = false;

  public:
    void setName(string value) { name = value; }
    bool available() const {
        if (suspended) return false;
        if (busy) return false;
        return true;
    }
    void setBusy(bool value) { busy = value; }
    void suspend(bool value) { suspended = value; }
    void show() const {
        cout << name;
        if (suspended) cout << "; отстранен";
        else cout << "; допущен";
        if (busy) cout << "; в рейсе\n";
        else cout << "; свободен\n";
    }
};

class Car {
  private:
    string number;
    // Даже свободную машину нельзя назначать, если ей нужен ремонт.
    bool repair = false;
    bool busy = false;

  public:
    void setNumber(string value) { number = value; }
    bool available() const {
        if (repair) return false;
        if (busy) return false;
        return true;
    }
    bool isBusy() const { return busy; }
    void setBusy(bool value) { busy = value; }
    void setRepair(bool value) { repair = value; }
    void show() const {
        cout << number;
        if (repair) cout << "; требуется ремонт";
        else cout << "; исправен";
        if (busy) cout << "; в рейсе\n";
        else cout << "; свободен\n";
    }
};

class Depot {
  private:
    class Trip {
        // Вложенный класс нужен только для рейсов этой автобазы.
      private:
        string route;
        // В рейсе храним индексы водителя и машины из общих массивов.
        int driver = 0, car = 0;
        bool done = false;

      public:
        void create(string r, int d, int c) {
            route = r;
            driver = d;
            car = c;
        }
        int driverIndex() const { return driver; }
        int carIndex() const { return car; }
        bool completed() const { return done; }
        void finish() { done = true; }
        void show() const {
            cout << route << "; водитель " << driver + 1 << "; авто "
                 << car + 1;
            if (done) cout << "; выполнен\n";
            else cout << "; выполняется\n";
        }
    };
    Driver *drivers;
    Car *cars; // Агрегация: эти массивы созданы снаружи автобазы.
    int driverCount = 0, carCount = 0, tripCount = 0;
    Trip trips[100]; // Композиция: рейсы принадлежат автобазе.
  public:
    // Указатели сохраняют адреса массивов, которые созданы в main.
    Depot(Driver d[], Car c[]) : drivers(d), cars(c) {}
    void addDriver() {
        // Проверяем место до записи нового человека в массив.
        if (driverCount == 100) {
            cout << "Список водителей заполнен.\n";
            return;
        }
        string name = readText("ФИО водителя: ");
        drivers[driverCount].setName(name);
        driverCount++;
    }
    void addCar() {
        if (carCount == 100) {
            cout << "Список автомобилей заполнен.\n";
            return;
        }
        string number = readText("Номер автомобиля: ", 20);
        cars[carCount].setNumber(number);
        carCount++;
    }
    void show() const {
        cout << "Водители:\n";
        // В массиве индексы с нуля, а пользователю показываем номера с единицы.
        for (int i = 0; i < driverCount; i++) {
            cout << i + 1 << ". ";
            drivers[i].show();
        }
        cout << "Автомобили:\n";
        for (int i = 0; i < carCount; i++) {
            cout << i + 1 << ". ";
            cars[i].show();
        }
        cout << "Рейсы:\n";
        for (int i = 0; i < tripCount; i++) {
            cout << i + 1 << ". ";
            trips[i].show();
        }
        if (!tripCount) cout << "Рейсов нет.\n";
    }
    void assign() {
        // Для нового рейса нужны записи в обоих списках и свободное место.
        if (!driverCount || !carCount || tripCount == 100) {
            cout << "Нужны водитель, автомобиль и свободное место в списке "
                    "рейсов.\n";
            return;
        }
        int d = readInt("Номер водителя: ", 1, driverCount) - 1;
        int c = readInt("Номер авто: ", 1, carCount) - 1;
        if (!drivers[d].available() || !cars[c].available()) {
            cout << "Водитель или автомобиль недоступен. Рейс не назначен.\n";
            return;
        }
        string route = readText("Маршрут заявки: ");
        trips[tripCount].create(route, d, c);
        tripCount++;
        // Один автомобиль и один водитель не могут выполнять два рейса сразу.
        drivers[d].setBusy(true);
        cars[c].setBusy(true);
        cout << "Рейс назначен.\n";
    }
    void finish() {
        if (tripCount == 0) {
            cout << "Рейсов нет.\n";
            return;
        }
        int t = readInt("Номер рейса: ", 1, tripCount) - 1;
        if (trips[t].completed()) {
            cout << "Рейс уже выполнен.\n";
            return;
        }
        int broken =
            readInt("Состояние авто: 0 - исправно, 1 - нужен ремонт: ", 0, 1);
        trips[t].finish();
        int d = trips[t].driverIndex();
        int c = trips[t].carIndex();
        // Завершение рейса одновременно освобождает человека и машину.
        drivers[d].setBusy(false);
        cars[c].setBusy(false);
        cars[c].setRepair(broken == 1);
        cout << "Рейс выполнен, состояние автомобиля сохранено.\n";
    }
    void repairRequest() {
        if (driverCount == 0 || carCount == 0) {
            cout << "Нет водителей или автомобилей.\n";
            return;
        }
        int d = readInt("Номер водителя: ", 1, driverCount) - 1;
        int c = readInt("Номер авто: ", 1, carCount) - 1;
        if (cars[c].isBusy()) {
            bool assigned = false;
            // Ищем действующий рейс с этой парой водителя и автомобиля.
            for (int i = 0; i < tripCount; i++) {
                if (trips[i].completed()) continue;
                if (trips[i].driverIndex() != d) continue;
                if (trips[i].carIndex() != c) continue;
                assigned = true;
            }
            if (!assigned) {
                cout << "Этот автомобиль закреплен за другим водителем.\n";
                return;
            }
            cout << "Сначала завершите рейс и укажите, что нужен ремонт.\n";
            return;
        }
        cars[c].setRepair(true);
        // Флаг ремонта сразу исключает машину из новых назначений.
        cout << "Заявка водителя на ремонт принята.\n";
    }
    void suspendDriver() {
        // Допуск влияет на будущие рейсы, а текущий остается в истории.
        if (driverCount == 0) {
            cout << "Водителей нет.\n";
            return;
        }
        int d = readInt("Номер водителя: ", 1, driverCount) - 1;
        int value = readInt("1 - отстранить, 0 - допустить: ", 0, 1);
        drivers[d].suspend(value == 1);
        cout << "Допуск изменен. Текущий рейс можно завершить; новый получит "
                "только допущенный водитель.\n";
    }
    void repairComplete() {
        // Отмечать ремонт завершенным можно только у свободной машины.
        if (carCount == 0) {
            cout << "Автомобилей нет.\n";
            return;
        }
        int c = readInt("Номер авто: ", 1, carCount) - 1;
        if (cars[c].isBusy()) {
            cout << "Нельзя ремонтировать автомобиль во время рейса.\n";
            return;
        }
        cars[c].setRepair(false);
        cout << "Автомобиль исправен.\n";
    }
};

int main() {
    // Массивы существуют все время работы, их адреса остаются действительными.
    Driver drivers[100];
    Car cars[100];
    Depot depot(drivers, cars);
    while (true) {
        cout << "\n1. Добавить водителя\n2. Добавить авто\n3. Списки\n4. "
                "Назначить рейс\n"
             << "5. Завершить рейс\n6. Заявка водителя на ремонт\n7. Допуск "
                "водителя\n8. Ремонт завершен\n0. Выход\n";
        int menu = readInt("Выбор: ", 0, 8);
        if (menu == 0) break;
        if (menu == 1) depot.addDriver();
        if (menu == 2) depot.addCar();
        if (menu == 3) depot.show();
        if (menu == 4) depot.assign();
        if (menu == 5) depot.finish();
        if (menu == 6) depot.repairRequest();
        if (menu == 7) depot.suspendDriver();
        if (menu == 8) depot.repairComplete();
    }
}
```

#ifndef INPUT_H
#define INPUT_H

#include <iostream>
#include <string>

// Здесь только ввод. Алгоритмы самих задач находятся в файлах taskN.cpp.
// inline позволяет хранить эти небольшие функции в общем заголовочном файле.
inline string readLine(string prompt) {
    cout << prompt;
    string line;
    if (!getline(cin, line)) {
        cout << "\nВвод завершён.\n";
        exit(0);
    }
    return line;
}

inline string readText(string prompt, size_t maxLength = 100) {
    while (true) {
        string text = readLine(prompt);
        if (text.size() > maxLength) {
            cout << "Ошибка: введите непустой текст (до " << maxLength
                 << " байт UTF-8).\n";
            continue;
        }
        // Убираем пробелы в начале и конце ответа.
        while (!text.empty() && (text.front() == ' ' || text.front() == '\t' ||
                                 text.front() == '\r')) {
            text.erase(0, 1);
        }
        while (!text.empty() && (text.back() == ' ' || text.back() == '\t' ||
                                 text.back() == '\r')) {
            text.pop_back();
        }

        bool valid = true;
        if (text.empty()) {
            valid = false;
        }
        for (unsigned char c : text) {
            // Служебные символы не должны ломать таблицу в консоли.
            if (c < 32 || c == 127) {
                valid = false;
            }
        }
        if (valid) {
            return text;
        }
        cout << "Ошибка: введите непустой текст (до " << maxLength
             << " байт UTF-8).\n";
    }
}

inline int readInt(string prompt, int low, int high) {
    while (true) {
        string line = readLine(prompt);
        // Читаем число из всей строки, чтобы не принять 12abc за число 12.
        istringstream input(line);
        int value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        if (value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно целое число от " << low << " до " << high
             << ".\n";
    }
}

inline double readDouble(string prompt, double low, double high) {
    while (true) {
        istringstream input(readLine(prompt));
        double value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        // isfinite отсеивает бесконечность и недопустимый результат NaN.
        if (!isfinite(value) || value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно конечное число от " << low << " до " << high
             << ". Дробная часть отделяется точкой.\n";
    }
}

inline bool validDate(string date) {
    if (date.size() != 10 || date[2] != '.' || date[5] != '.') {
        return false;
    }
    for (int i = 0; i < 10; i++) {
        if (i != 2 && i != 5 && (date[i] < '0' || date[i] > '9')) {
            return false;
        }
    }
    // substr берёт часть строки, stoi переводит её в целое число.
    int day = stoi(date.substr(0, 2));
    int month = stoi(date.substr(3, 2));
    int year = stoi(date.substr(6, 4));
    if (year < 1900 || year > 2100 || month < 1 || month > 12) {
        return false;
    }
    int days[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    // Високосный год добавляет один день в феврале.
    if (year % 400 == 0 || (year % 4 == 0 && year % 100 != 0)) {
        days[1] = 29;
    }
    return day >= 1 && day <= days[month - 1];
}

// Запись ГГГГММДД позволяет правильно сравнить две даты как числа.
inline int dateNumber(string date) {
    return stoi(date.substr(6, 4)) * 10000 + stoi(date.substr(3, 2)) * 100 +
           stoi(date.substr(0, 2));
}

inline string readDate(string prompt) {
    while (true) {
        string date = readText(prompt, 10);
        if (validDate(date)) {
            return date;
        }
        cout << "Ошибка: нужна существующая дата ДД.ММ.ГГГГ, год 1900..2100.\n";
    }
}

inline string readTime(string prompt) {
    while (true) {
        string time = readText(prompt, 5);
        bool valid = time.size() == 5;
        if (valid) {
            valid = time[2] == ':';
            for (int i = 0; i < 5; i++) {
                if (i != 2 && (time[i] < '0' || time[i] > '9')) {
                    valid = false;
                }
            }
        }
        if (valid) {
            int hours = stoi(time.substr(0, 2));
            int minutes = stoi(time.substr(3, 2));
            if (hours <= 23 && minutes <= 59) {
                return time;
            }
        }
        cout << "Ошибка: время должно быть от 00:00 до 23:59.\n";
    }
}

inline string readPhone(string prompt) {
    while (true) {
        string phone = readText(prompt, 16);
        size_t start = 0;
        if (phone[0] == '+') {
            start = 1;
        }
        bool valid = phone.size() - start >= 7 && phone.size() - start <= 15;
        for (size_t i = start; i < phone.size(); i++) {
            if (phone[i] < '0' || phone[i] > '9') {
                valid = false;
            }
        }
        if (valid) {
            return phone;
        }
        cout << "Ошибка: нужно 7..15 цифр; знак + допустим только в начале.\n";
    }
}
