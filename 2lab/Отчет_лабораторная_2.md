### 3.1 Таблица рейсов Aeroflot

Условие задачи. Создать класс Aeroflot с закрытыми полями: номер рейса, пункты отправления и назначения, время прибытия и отправления, место регистрации. Все методы определить внутри класса. Через меню вводить рейсы, выводить таблицу, сортировать по каждому полю и искать по пункту назначения. В main использовать массив объектов.

Данные программы. number — номер рейса, строка; departure и destination — названия пунктов; arrivalTime и departureTime — время ЧЧ:ММ; registration — место регистрации. input вводит поля; show выводит запись; field выбирает поле; showAll, sort и search обрабатывают массив объектов. Статические методы получают массив явно и не требуют отдельного объекта.

Проверки и ограничения. До 100 рейсов. Время 00:00..23:59. Даты и часовые пояса не заданы, поэтому прибытие раньше отправления по часам допустимо. Номер хранится строкой (например, AB12); повторы разрешены, так как дата вылета не задана. Поиск точный, с учётом регистра. Сортировка идёт по кодам UTF-8 без особого словарного порядка для Ё.

#### 3.1.1 Алгоритм

1. Создать массив на 100 рейсов; обнулить счётчик.

2. Показать меню и проверить команду.

3. Добавить рейс, если есть место; проверить все поля.

4. Вывести сохранённые рейсы в цикле.

5. Сортировать по выбранному полю обменом соседних записей.

6. Найти все рейсы с введённым пунктом назначения.

7. Повторить меню или выйти при команде 0.

Входные данные. Пункт меню; шесть полей рейса; поле сортировки; пункт назначения для поиска.

Выходные данные. Таблица рейсов, найденные записи и сообщения об ошибках или отсутствии записей.

```cpp
#include "../common/input.h" //Подключаем сторонний заголовочный файл input.h, который лежит на папку выше (../) в папке common. В нём объявлены вспомогательные функции для безопасного ввода данных (например, readText, readTime, readInt).

// Один объект Aeroflot хранит один рейс.
// Все рейсы вместе хранятся в обычном массиве в main.
class Aeroflot { //чертежа (класса)
private:
    // private запрещает менять поля напрямую из main.
    string number, departure, destination, arrivalTime, departureTime, registration;

public:
    // Методы внутри класса являются встроенными (inline).
    void input() {
        // readText и readTime повторяют запрос, если ответ неверный.
        number = readText("Номер рейса: ", 20);
        departure = readText("Пункт отправления: ", 60);
        destination = readText("Пункт назначения: ", 60);
        arrivalTime = readTime("Время прибытия ЧЧ:ММ: ");
        departureTime = readTime("Время отправления ЧЧ:ММ: ");
        registration = readText("Место регистрации: ", 30);
    }

    // const: этот метод только показывает данные и не меняет их.
    void show() const { 
        cout << number << " | " << departure << " | " << destination << " | "
             << arrivalTime << " | " << departureTime << " | " << registration << '\n';
    }

    string field(int choice) const {
        // Возвращаем поле, которое пользователь выбрал для сортировки по индексу.
        if (choice == 1) return number; //Если передали 1 — возвращаем номер рейса.
        if (choice == 2) return departure;
        if (choice == 3) return destination;
        if (choice == 4) return arrivalTime;
        if (choice == 5) return departureTime;
        return registration;
    }

    static void showAll(const Aeroflot flights[], int count) {
        // static: метод работает с переданным массивом, а не с одним рейсом.
        if (count == 0) {
            cout << "Рейсов нет.\n";
            return;
        }
        cout << "Рейс | Откуда | Куда | Прибытие | Отправление | Регистрация\n";
        for (int i = 0; i < count; i++) {
            flights[i].show();
        }
    }

    static void sort(Aeroflot flights[], int count, int choice) {
        // Пузырьковая сортировка меняет местами соседние записи.
        for (int i = 0; i < count - 1; i++) { 
            // После каждого прохода один элемент уже стоит в конце правильно.
            for (int j = 0; j < count - 1 - i; j++) { //Внутренний цикл: сравнивает соседние пары элементов. С каждым шагом i правая часть массива уже отсортирована, поэтому вычитаем i
                if (flights[j].field(choice) > flights[j + 1].field(choice)) { //Сравниваем значение выбранного поля у левого элемента (j) и правого (j + 1). Если левое "больше" (по алфавиту/строково), их нужно поменять местами.
                    Aeroflot temp = flights[j]; //Создаём временный объект temp и сохраняем в него левый рейс.
                    flights[j] = flights[j + 1]; //Перезаписываем левый рейс правым
                    flights[j + 1] = temp; //В правый рейс записываем сохранённый из temp левый.
                }
            }
        }
        showAll(flights, count);
    }

    static void search(const Aeroflot flights[], int count) {
        string place = readText("Пункт назначения для поиска: ", 60);
        bool found = false;
        // Просматриваем все рейсы: совпадений может быть несколько.
        for (int i = 0; i < count; i++) {
            if (flights[i].destination == place) {
                flights[i].show();
                found = true;
            }
        }
        if (!found) {
            cout << "Рейс не найден.\n";
        }
    }
};

int main() {
    const int MAX = 100;
    Aeroflot flights[MAX];
    int count = 0; // Сколько ячеек массива уже занято.
    // После каждой операции снова показываем меню.
    while (true) {
        cout << "\n1 - добавить рейс, 2 - таблица, 3 - сортировка, 4 - поиск, 0 - выход\n";
        int menu = readInt("Выбор: ", 0, 4);
        // Для каждого пункта меню вызываем нужный метод класса.
        if (menu == 0) {
            break;
        }
        if (menu == 1) {
            if (count == MAX) {
                cout << "Достигнут предел: 100 рейсов.\n";
            } else {
                flights[count].input();
                count++;
            }
        }
        if (menu == 2) {
            Aeroflot::showAll(flights, count);
        }
        if (menu == 3) {
            cout << "1 - номер, 2 - отправление, 3 - назначение, 4 - прибытие,\n"
                 << "5 - время отправления, 6 - регистрация\n";
            int field = readInt("Поле: ", 1, 6);
            Aeroflot::sort(flights, count, field);
        }
        if (menu == 4) {
            Aeroflot::search(flights, count);
        }
    }
}
```

### 3.2 Студенческая группа studGroup

Условие задачи. Описать класс studGroup с закрытыми именем, фамилией, отчеством, датой рождения и телефоном. Работать с переменным числом объектов в массиве. Через меню добавлять, удалять, выводить, искать и сортировать по каждому полю. Поиск и сортировку определить вне класса, остальные методы — внутри.

Данные программы. name, surname, patronymic, birthDate, phone — закрытые строки. Счётчик count задаёт фактическое число студентов; вместимость массива — 100. remove сдвигает записи; sort сравнивает даты численно в порядке ГГГГММДД; search выводит все совпадения.

Проверки и ограничения. До 100 студентов. Для отсутствующего отчества вводится знак -. Даты 1900..2100, дата рождения не позже текущей даты компьютера. Телефон: 7..15 цифр и необязательный начальный плюс. Это проверка формата, а не существования номера. Текстовый поиск учитывает регистр. Сортировка телефона лексикографическая; сортировка даты календарная.

#### 3.2.1 Алгоритм

1. Создать пустой массив записей и счётчик.

2. Прочитать проверенный пункт меню.

3. При добавлении ввести ФИО, проверить календарную дату, отсутствие будущей даты и формат телефона.

4. При удалении проверить номер, сдвинуть последующие записи влево и уменьшить счётчик.

5. При сортировке выбрать поле; даты сравнивать хронологически.

6. При поиске проверить запрос и вывести все совпадения или сообщение об отсутствии.

7. Вернуться в меню или завершить программу.

#### 3.2.3 Программный код

```cpp
#include "../common/input.h"
#include <ctime>

// Один объект studGroup описывает одного студента.
// Методы поиска и сортировки ниже вынесены за пределы класса.
class studGroup {
  private:
    string name, surname, patronymic, birthDate, phone;

  public:
    void input() {
        // Поля заполняются через метод, так как снаружи они закрыты.
        name = readText("Имя: ", 60);
        surname = readText("Фамилия: ", 60);
        patronymic = readText("Отчество (или -): ", 60);
        // Берём текущую дату, чтобы не принять будущую дату рождения.
        time_t now = time(nullptr);
        tm *today = localtime(&now);
        int todayNumber = 21001231;
        if (today != nullptr) {
            int year = today->tm_year + 1900;
            int month = today->tm_mon + 1;
            todayNumber = year * 10000 + month * 100 + today->tm_mday;
        }
        do {
            birthDate = readDate("Дата рождения ДД.ММ.ГГГГ: ");
            if (dateNumber(birthDate) > todayNumber) {
                cout << "Дата рождения не может быть в будущем.\n";
            }
        } while (dateNumber(birthDate) > todayNumber);
        phone = readPhone("Телефон: ");
    }

    void show() const {
        cout << surname << " | " << name << " | " << patronymic << " | "
             << birthDate << " | " << phone << '\n';
    }

    string field(int choice) const {
        // Номер поля берём из меню поиска или сортировки.
        if (choice == 1) return name;
        if (choice == 2) return surname;
        if (choice == 3) return patronymic;
        if (choice == 4) return birthDate;
        return phone;
    }

    static void showAll(const studGroup group[], int count) {
        if (count == 0) {
            cout << "Группа пуста.\n";
            return;
        }
        cout << "№ | Фамилия | Имя | Отчество | Дата рождения | Телефон\n";
        for (int i = 0; i < count; i++) {
            cout << i + 1 << " | ";
            group[i].show();
        }
    }

    static void remove(studGroup group[], int &count, int index) {
        // Сдвиг записей закрывает место удалённого студента.
        for (int i = index; i < count - 1; i++) {
            group[i] = group[i + 1];
        }
        count--;
    }

    static void sort(studGroup group[], int count, int choice);
    static void search(const studGroup group[], int count, int choice);
};

// По условию поиск и сортировка определены вне класса.
void studGroup::sort(studGroup group[], int count, int choice) {
    // За один проход самая большая запись перемещается в конец.
    for (int i = 0; i < count - 1; i++) {
        for (int j = 0; j < count - 1 - i; j++) {
            bool swapNeeded;
            if (choice == 4) {
                // Дату сравниваем по году, месяцу и дню, а не как текст.
                int firstDate = dateNumber(group[j].birthDate);
                int secondDate = dateNumber(group[j + 1].birthDate);
                swapNeeded = firstDate > secondDate;
            } else {
                swapNeeded =
                    group[j].field(choice) > group[j + 1].field(choice);
            }
            if (swapNeeded) {
                studGroup temp = group[j];
                group[j] = group[j + 1];
                group[j + 1] = temp;
            }
        }
    }
    showAll(group, count);
}

void studGroup::search(const studGroup group[], int count, int choice) {
    string value;
    if (choice == 4) {
        value = readDate("Дата: ");
    } else if (choice == 5) {
```

### 3.3 Сортировочный узел на основе стека

Условие задачи. Описать класс Stack на указателях. Сформировать состав из вагонов двух типов с клавиатуры или из файла, показать его до разделения, распределить вагоны в два стека по типам, затем показать исходный и полученные составы.

Данные программы. Node содержит тип вагона и указатель next; top указывает на вершину стека. push добавляет узел; pop извлекает значение и удаляет узел; clear очищает стек; divide разделяет состав. Входной файл: число вагонов N, затем ровно N чисел 1 или 2, разделённых пробелами или переводами строк. Пример — train.txt.

Проверки и ограничения. Вершина — последний введённый вагон. В программе показывается именно порядок от вершины к основанию. Неверный или неполный файл не заменяет уже введённый состав. Лишние данные в файле запрещены. Тип 1 отправляется в направление 1; тип 2 — в направление 2. Повторное разделение пустого состава создаёт два пустых результата.

#### 3.3.1 Алгоритм

1. Выбрать ввод состава с клавиатуры или из файла.

2. Проверить количество 0..100 и тип каждого вагона 1 или 2.

3. После полной проверки заменить исходный состав, помещая вагоны в стек.

4. Вывести исходный стек от вершины.

5. Пока исходный стек не пуст, снять верхний вагон и поместить его в стек соответствующего направления.

6. Вывести пустой исходный стек и два полученных состава.

7. При завершении освободить узлы стеков.

```cpp
#include "../common/input.h"
#include <fstream>

// Стек: последним добавили вагон — первым его забрали.
// Указатель next связывает один вагон со следующим.
class Stack {
  private:
    struct Node {
        int wagon;
        Node *next;
    };
    Node *top = nullptr;

  public:
    void push(int wagon) {
        Node *node = new Node;
        node->wagon = wagon;
        node->next = top; // Новый узел указывает на прежнюю вершину.
        top = node;
    }

    bool pop(int &wagon) {
        // Ссылка & позволяет вернуть снятый вагон через параметр функции.
        if (top == nullptr) {
            return false;
        }
        Node *old = top;
        wagon = top->wagon;
        top = top->next;
        delete old; // Освобождаем память удалённого узла.
        return true;
    }

    void clear() {
        int wagon;
        // pop сам удаляет узел. Повторяем, пока стек не станет пустым.
        while (pop(wagon)) {
        }
    }

    void show() const {
        cout << "Вершина -> ";
        Node *current = top;
        while (current != nullptr) {
            cout << current->wagon << ' ';
            current = current->next;
        }
        if (top == nullptr) {
            cout << "пусто";
        }
        cout << '\n';
    }

    bool input() {
        // Временный массив нужен, чтобы ошибка файла не испортила старый стек.
        int wagons[100], count;
        int mode = readInt("1 - клавиатура, 2 - файл: ", 1, 2);
        if (mode == 1) {
            count = readInt("Количество вагонов (0..100): ", 0, 100);
            for (int i = 0; i < count; i++) {
                wagons[i] = readInt("Тип вагона (1 или 2): ", 1, 2);
            }
        } else {
            string path = readText("Путь к файлу: ", 250);
            ifstream file(path);
            // Сначала читаем количество, затем ровно столько вагонов.
            if (!file || !(file >> count) || count < 0 || count > 100) {
                cout << "Ошибка открытия файла или количества вагонов.\n";
                return false;
            }
            for (int i = 0; i < count; i++) {
                if (!(file >> wagons[i]) ||
                    (wagons[i] != 1 && wagons[i] != 2)) {
                    cout << "Ошибка: каждый вагон должен иметь тип 1 или 2.\n";
                    return false;
                }
            }
            string extra;
            if (file >> extra) {
                cout << "Ошибка: лишние данные в файле.\n";
                return false;
            }
        }
        clear(); // Старый состав заменяется только после проверки всех данных.
        for (int i = 0; i < count; i++) {
            push(wagons[i]);
        }
        return true;
    }

    void divide(Stack &first, Stack &second) {
        // Перед новым разделением очищаем предыдущие результаты.
        first.clear();
        second.clear();
        int wagon;
        while (pop(wagon)) {
            // Тип вагона определяет направление.
            if (wagon == 1) {
                first.push(wagon);
            } else {
                second.push(wagon);
            }
        }
    }

    // При обычном выходе из main освобождаем оставшиеся узлы.
    ~Stack() { clear(); }
};

int main() {
    Stack train, first, second;
    while (true) {
        cout << "\n1 - сформировать состав, 2 - показать, 3 - разделить, 0 - "
                "выход\n";
        int menu = readInt("Выбор: ", 0, 3);
        if (menu == 0) {
            break;
        }
        if (menu == 1) {
            bool loaded = train.input();
            if (loaded) {
                first.clear();
                second.clear();
            }
        }
        if (menu == 2) {
            cout << "Общий: ";
            train.show();
        }
        if (menu == 3) {
            cout << "До разделения: ";
            train.show();
            train.divide(first, second);
            cout << "Общий после разделения: ";
            train.show();
            cout << "Направление 1: ";
            first.show();
            cout << "Направление 2: ";
            second.show();
        }
    }
}
```

```cpp
#ifndef INPUT_H

// inline позволяет хранить эти небольшие функции в общем заголовочном файле.
inline string readLine(string prompt) {
    cout << prompt;
    string line;
    if (!getline(cin, line)) {
        cout << "\nВвод завершён.\n";
        exit(0);
    }
    return line;
}

inline string readText(string prompt, size_t maxLength = 100) {
    while (true) {
        string text = readLine(prompt);
        if (text.size() > maxLength) {
            cout << "Ошибка: введите непустой текст (до " << maxLength
                 << " байт UTF-8).\n";
            continue;
        }
        // Убираем пробелы в начале и конце ответа.
        while (!text.empty() && (text.front() == ' ' || text.front() == '\t' ||
                                 text.front() == '\r')) {
            text.erase(0, 1);
        }
        while (!text.empty() && (text.back() == ' ' || text.back() == '\t' ||
                                 text.back() == '\r')) {
            text.pop_back();
        }

        bool valid = true;
        if (text.empty()) {
            valid = false;
        }
        for (unsigned char c : text) {
            // Служебные символы не должны ломать таблицу в консоли.
            if (c < 32 || c == 127) {
                valid = false;
            }
        }
        if (valid) {
            return text;
        }
        cout << "Ошибка: введите непустой текст (до " << maxLength
             << " байт UTF-8).\n";
    }
}

inline int readInt(string prompt, int low, int high) {
    while (true) {
        string line = readLine(prompt);
        // Читаем число из всей строки, чтобы не принять 12abc за число 12.
        istringstream input(line);
        int value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        if (value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно целое число от " << low << " до " << high
             << ".\n";
    }
}

inline double readDouble(string prompt, double low, double high) {
    while (true) {
        istringstream input(readLine(prompt));
        double value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        // isfinite отсеивает бесконечность и недопустимый результат NaN.
        if (!isfinite(value) || value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно конечное число от " << low << " до " << high
             << ". Дробная часть отделяется точкой.\n";
    }
}

inline bool validDate(string date) {
    if (date.size() != 10 || date[2] != '.' || date[5] != '.') {
        return false;
    }
    for (int i = 0; i < 10; i++) {
        if (i != 2 && i != 5 && (date[i] < '0' || date[i] > '9')) {
            return false;
        }
    }
    // substr берёт часть строки, stoi переводит её в целое число.
    int day = stoi(date.substr(0, 2));
    int month = stoi(date.substr(3, 2));
    int year = stoi(date.substr(6, 4));
    if (year < 1900 || year > 2100 || month < 1 || month > 12) {
        return false;
    }
    int days[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    // Високосный год добавляет один день в феврале.
    if (year % 400 == 0 || (year % 4 == 0 && year % 100 != 0)) {
        days[1] = 29;
    }
    return day >= 1 && day <= days[month - 1];
}

inline int dateNumber(string date) {
    return stoi(date.substr(6, 4)) * 10000 + stoi(date.substr(3, 2)) * 100 +
           stoi(date.substr(0, 2));
}

inline string readDate(string prompt) {
    while (true) {
        string date = readText(prompt, 10);
        if (validDate(date)) {
            return date;
        }
        cout << "Ошибка: нужна существующая дата ДД.ММ.ГГГГ, год 1900..2100.\n";
    }
}

inline string readTime(string prompt) {
    while (true) {
        string time = readText(prompt, 5);
        bool valid = time.size() == 5;
        if (valid) {
            valid = time[2] == ':';
            for (int i = 0; i < 5; i++) {
                if (i != 2 && (time[i] < '0' || time[i] > '9')) {
                    valid = false;
                }
            }
        }
        if (valid) {
            int hours = stoi(time.substr(0, 2));
            int minutes = stoi(time.substr(3, 2));
            if (hours <= 23 && minutes <= 59) {
                return time;
            }
        }
        cout << "Ошибка: время должно быть от 00:00 до 23:59.\n";
    }
}

inline string readPhone(string prompt) {
    while (true) {
        string phone = readText(prompt, 16);
        size_t start = 0;
        if (phone[0] == '+') {
            start = 1;
        }
        bool valid = phone.size() - start >= 7 && phone.size() - start <= 15;
        for (size_t i = start; i < phone.size(); i++) {
            if (phone[i] < '0' || phone[i] > '9') {
                valid = false;
            }
        }
        if (valid) {
            return phone;
        }
        cout << "Ошибка: нужно 7..15 цифр; знак + допустим только в начале.\n";
    }
}
```
