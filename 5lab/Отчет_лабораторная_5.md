### 3.1 Класс Int с проверкой переполнения

Условие задачи. Создать класс Int и перегрузить внутренними операторными функциями +, -, *, /, префиксный и постфиксный ++. Промежуточные вычисления выполнять в long double, переполнение обнаруживать исключением с сообщением и завершением программы. Для унарных операторов использовать this.

Данные программы. value — закрытое целочисленное значение типа int. checked — общий закрытый метод проверки результата.

Проверки и ограничения. На проверенной 32-битной модели int диапазон -2147483648..2147483647. Верхняя граница 2147483648 в методичке исправлена как опечатка. Деление целочисленное: дробная часть отбрасывается к нулю. Проверка выполняется до преобразования, включая INT_MIN / -1. Текст, дробное число вместо int, хвост и число за пределами int запрашиваются повторно. EOF завершает ввод. Переполнение и нулевой делитель завершают программу с кодом 1.

#### 3.1.1 Алгоритм

1. Выбрать операцию меню и получить корректный int; для бинарной операции получить второй операнд.

2. Преобразовать левое значение в long double до арифметики. При делении сначала проверить нулевой делитель.

3. Сравнить промежуточный результат с numeric_limits<int>::min()/max(); при выходе за пределы бросить overflow_error.

4. При допустимом результате вернуть объект Int. Префикс изменяет текущий объект через this; постфикс сначала сохраняет его копию.

5. Вывести возвращённое значение и новое a для инкремента. Исключение перехватить, вывести причину и завершить с кодом 1.

#### 3.1.3 Программный код

```cpp
#include "../common/input.h"
#include <limits>
#include <stdexcept>

class Int {
    int value; // Значение объекта закрыто от прямого изменения.

    // static: эта проверка общая для всех объектов Int.
    static int checked(long double result) {
        // Проверяем границы до преобразования обратно в int.
        if (result < numeric_limits<int>::min() ||
            result > numeric_limits<int>::max())
            throw overflow_error("Переполнение типа int.");
        return static_cast<int>(result);
    }

  public:
    Int(int number = 0) : value(number) {}
    // const: чтение значения не меняет объект.
    int getValue() const { return value; }

    // const Int& передаёт второй объект без копирования и без изменения.
    Int operator+(const Int &other) const {
        // static_cast меняет тип до сложения, чтобы int не переполнился.
        return Int(checked(static_cast<long double>(value) + other.value));
    }
    Int operator-(const Int &other) const {
        return Int(checked(static_cast<long double>(value) - other.value));
    }
    Int operator*(const Int &other) const {
        return Int(checked(static_cast<long double>(value) * other.value));
    }
    Int operator/(const Int &other) const {
        // Нулевой делитель проверяем до самой операции.
        if (other.value == 0) throw domain_error("Деление на ноль.");
        return Int(checked(static_cast<long double>(value) / other.value));
    }
    Int &operator++() {
        this->value = checked(static_cast<long double>(value) + 1);
        return *this; // Префикс возвращает уже изменённый объект.
    }
    Int operator++(int) {
        Int old = *this;
        ++(*this);
        return old; // Постфикс возвращает значение до изменения.
    }
};

int main() {
    cout << "Int: диапазон " << numeric_limits<int>::min() << ".."
         << numeric_limits<int>::max() << ". Деление целочисленное.\n";
    try {
        // Цикл позволяет проверить разные операции за один запуск.
        while (true) {
            cout << "\n1 +  2 -  3 *  4 /  5 ++a  6 a++  0 Выход\n";
            int choice = readInt("Команда: ", 0, 6);
            if (choice == 0) break;
            // readInt повторяет ввод, если введены буквы или число вне границ.
            Int a(readInt("a: ", numeric_limits<int>::min(),
                          numeric_limits<int>::max()));
            Int result;
            if (choice <= 4) {
                Int b(readInt("b: ", numeric_limits<int>::min(),
                              numeric_limits<int>::max()));
                if (choice == 1) result = a + b;
                if (choice == 2) result = a - b;
                if (choice == 3) result = a * b;
                if (choice == 4) result = a / b;
            } else if (choice == 5) {
                result = ++a;
            } else {
                result = a++;
            }
            cout << "Результат: " << result.getValue() << '\n';
            if (choice >= 5)
                cout << "a после операции: " << a.getValue() << '\n';
        }
    } catch (const exception &error) {
        cout << "Ошибка: " << error.what() << " Программа завершена.\n";
        return 1;
    }
    return 0;
}
```

### 3.2 Класс fraction и операции с числами

Условие задачи. Класс fraction содержит одно закрытое поле chislo типа double. Требуются конструкторы по умолчанию и с одним аргументом, все четыре арифметические операции для объект/объект, объект/число, число/объект, инкремент и меню проверки методов.

Данные программы. chislo — закрытое вещественное значение double. Название поля сохранено из задания; слово «целочисленное» при double в методичке является опечаткой.

Проверки и ограничения. Ввод: конечные значения double с точкой как разделителем; дробь 3/4 вводится как 0.75. Хранение приближённое, как у double, без отдельных числителя и знаменателя. Деление на +0 и -0 запрещено. Переполнение и исчезновение ненулевого результата до нуля сообщаются как ошибки. NaN, inf и лишние символы ввода отвергаются. Ошибки арифметики не прерывают меню. Для числа слева используются внешние операторы; для вычитания и деления сохраняется порядок операндов.

#### 3.2.1 Алгоритм

1. Показать нулевое значение объекта, созданного конструктором по умолчанию. Выбрать операцию и при необходимости расположение объектов и чисел.

2. Считать конечные числа и создать объекты конструктором с аргументом. Выбрать нужную перегрузку operator.

3. Проверить деление на ноль; вычислить результат в long double, проверить диапазон double и потерю ненулевого значения до нуля.

4. При инкременте использовать *this; префикс возвращает новое значение, постфикс возвращает заранее сохранённое старое.

5. Вывести результат; при исключении вывести сообщение и продолжить меню.

#### 3.2.2 Блок-схема

#### 3.2.3 Программный код

```cpp
#include "../common/input.h"
#include <limits>
#include <stdexcept>

class fraction {
    double chislo; // Дробь хранится как обычное десятичное число.

    // static: одна проверка результатов для всех объектов класса.
    static fraction checked(long double result) {
        // Не допускаем бесконечность и потерю ненулевого результата до нуля.
        if (!isfinite(result) || fabs(result) > numeric_limits<double>::max())
            throw overflow_error("Результат слишком велик для double.");
        double value = static_cast<double>(result); // Возвращаемся к типу поля.
        if (result != 0 && value == 0)
            throw underflow_error("Результат слишком мал для double.");
        return fraction(value);
    }

  public:
    fraction() : chislo(0) {}
    fraction(double value) : chislo(value) {
        if (!isfinite(value)) throw domain_error("Число должно быть конечным.");
    }
    double getValue() const { return chislo; } // const запрещает менять объект.

    // const-ссылка позволяет читать второй объект, не создавая его копию.
    fraction operator+(const fraction &other) const {
        // long double даёт место для проверки слишком большого результата.
        return checked(static_cast<long double>(chislo) + other.chislo);
    }
    fraction operator-(const fraction &other) const {
        return checked(static_cast<long double>(chislo) - other.chislo);
    }
    fraction operator*(const fraction &other) const {
        return checked(static_cast<long double>(chislo) * other.chislo);
    }
    fraction operator/(const fraction &other) const {
        if (other.chislo == 0) throw domain_error("Деление на ноль.");
        return checked(static_cast<long double>(chislo) / other.chislo);
    }

    // Число справа превращается в fraction конструктором с одним аргументом.

    // Слева стоит число: внешний оператор сохраняет порядок операндов.
    friend fraction operator+(double number, const fraction &value) {
        return fraction(number) + value;
    }
    friend fraction operator-(double number, const fraction &value) {
        return fraction(number) - value;
    }
    friend fraction operator*(double number, const fraction &value) {
        return fraction(number) * value;
    }
    friend fraction operator/(double number, const fraction &value) {
        return fraction(number) / value;
    }

    fraction &operator++() {
        // this указывает на текущий объект; *this — сам этот объект.
        *this = *this + 1.0;
        return *this;
    }
    fraction operator++(int) {
        fraction old = *this; // Сохраняем прежнее значение для постфикса.
        ++(*this);
        return old;
    }
};

int main() {
    fraction zero;
    cout << setprecision(15) << "Конструктор по умолчанию: " << zero.getValue()
         << "\nДроби вводите десятичными числами, например 0.75 вместо 3/4.\n";
    double maximum = numeric_limits<double>::max();
    while (true) {
        cout << "\n1 +  2 -  3 *  4 /  5 ++a  6 a++  0 Выход\n";
        int choice = readInt("Команда: ", 0, 6);
        if (choice == 0) break;
        int form = 1;
        if (choice <= 4) {
            // Проверяем одинаковую операцию с тремя видами операндов.
            cout << "1 объект/объект; 2 объект/число; 3 число/объект\n";
            form = readInt("Вид операндов: ", 1, 3);
        }
        double left = readDouble("Первое значение: ", -maximum, maximum);
        fraction a(left), result;
        try {
            if (choice >= 5) {
                if (choice == 5) result = ++a;
                else result = a++;
                cout << "a после операции: " << a.getValue() << '\n';
            } else {
                double right =
                    readDouble("Второе значение: ", -maximum, maximum);
                fraction b(right);
                if (form == 1) {
                    if (choice == 1) result = a + b;
                    if (choice == 2) result = a - b;
                    if (choice == 3) result = a * b;
                    if (choice == 4) result = a / b;
                } else if (form == 2) {
                    if (choice == 1) result = a + right;
                    if (choice == 2) result = a - right;
                    if (choice == 3) result = a * right;
                    if (choice == 4) result = a / right;
                } else {
                    if (choice == 1) result = left + b;
                    if (choice == 2) result = left - b;
                    if (choice == 3) result = left * b;
                    if (choice == 4) result = left / b;
                }
            }
            cout << "Результат: " << result.getValue() << '\n';
        } catch (const exception &error) {
            // Ошибка операции не мешает выбрать следующую команду меню.
            cout << "Ошибка: " << error.what() << '\n';
        }
    }
    return 0;
}
```

### 3.3 Перегрузка метода rect_area

Условие задачи. Создать две версии метода rect_area: с двумя сторонами для прямоугольника и с одной стороной для квадрата. Показать выбор перегрузки по числу аргументов.

Данные программы. Класс Figure не хранит поля: стороны передаются параметрами метода, площадь возвращается как double.

Проверки и ограничения. Каждая сторона ограничена диапазоном 0.000001..1000000. Поэтому площадь лежит от 1e-12 до 1e12, без переполнения или исчезновения в double. Неположительные, нечисловые, бесконечные и выходящие за диапазон размеры запрашиваются повторно. В этой задаче исключения не используются. EOF завершает ввод.

#### 3.3.1 Алгоритм

1. Выбрать прямоугольник или квадрат в меню.

2. Считать одну положительную сторону, а для прямоугольника — вторую.

3. Вызвать rect_area(a, b) или rect_area(a). Вычислить произведение сторон либо квадрат стороны.

4. Вывести площадь и вернуться в меню.

#### 3.3.3 Программный код

```cpp
#include "../common/input.h"

class Figure {
  public:
    // Два аргумента означают длину и ширину прямоугольника.
    double rect_area(double width, double height) const {
        return width * height;
    }
    double rect_area(double side) const {
        return side * side; // Для квадрата достаточно одного размера.
    }
};

int main() {
    Figure figure;
    // После расчёта можно выбрать следующую фигуру.
    while (true) {
        cout << "\n1 Прямоугольник  2 Квадрат  0 Выход\n";
        int choice = readInt("Команда: ", 0, 2);
        if (choice == 0) break;
        // Этот диапазон сторон исключает переполнение при умножении.
        double a =
            readDouble("Сторона a (0.000001..1000000): ", 0.000001, 1000000);
        double area;
        if (choice == 1) {
            double b = readDouble("Сторона b (0.000001..1000000): ", 0.000001,
                                  1000000);
            area = figure.rect_area(a, b);
        } else {
            area = figure.rect_area(a);
        }
        cout << setprecision(15) << "Площадь: " << area << '\n';
    }
    return 0;
}
```

```cpp
        }
        // Убираем пробелы в начале и конце ответа.
        while (!text.empty() && (text.front() == ' ' || text.front() == '\t' ||
                                 text.front() == '\r')) {
            text.erase(0, 1);
        }
        while (!text.empty() && (text.back() == ' ' || text.back() == '\t' ||
                                 text.back() == '\r')) {
            text.pop_back();
        }

        bool valid = true;
        if (text.empty()) {
            valid = false;
        }
        for (unsigned char c : text) {
            // Служебные символы не должны ломать таблицу в консоли.
            if (c < 32 || c == 127) {
                valid = false;
            }
        }
        if (valid) {
            return text;
        }
        cout << "Ошибка: введите непустой текст (до " << maxLength
             << " байт UTF-8).\n";
    }
}

inline int readInt(string prompt, int low, int high) {
    while (true) {
        string line = readLine(prompt);
        // Читаем число из всей строки, чтобы не принять 12abc за число 12.
        istringstream input(line);
        int value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        if (value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно целое число от " << low << " до " << high
             << ".\n";
    }
}

inline double readDouble(string prompt, double low, double high) {
    while (true) {
        istringstream input(readLine(prompt));
        double value = 0;
        char extra;
        input >> value;
        bool valid = !input.fail();
        if (input >> extra) {
            valid = false;
        }
        // isfinite отсеивает бесконечность и недопустимый результат NaN.
        if (!isfinite(value) || value < low || value > high) {
            valid = false;
        }
        if (valid) {
            return value;
        }
        cout << "Ошибка: нужно конечное число от " << low << " до " << high
             << ". Дробная часть отделяется точкой.\n";
    }
}

inline bool validDate(string date) {
    if (date.size() != 10 || date[2] != '.' || date[5] != '.') {
        return false;
    }
    for (int i = 0; i < 10; i++) {
        if (i != 2 && i != 5 && (date[i] < '0' || date[i] > '9')) {
            return false;
        }
    }
    // substr берёт часть строки, stoi переводит её в целое число.
    int day = stoi(date.substr(0, 2));
    int month = stoi(date.substr(3, 2));
    int year = stoi(date.substr(6, 4));
    if (year < 1900 || year > 2100 || month < 1 || month > 12) {
        return false;
    }
    int days[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    // Високосный год добавляет один день в феврале.
    if (year % 400 == 0 || (year % 4 == 0 && year % 100 != 0)) {
        days[1] = 29;
    }
    return day >= 1 && day <= days[month - 1];
}

// Запись ГГГГММДД позволяет правильно сравнить две даты как числа.
inline int dateNumber(string date) {
    return stoi(date.substr(6, 4)) * 10000 + stoi(date.substr(3, 2)) * 100 +
           stoi(date.substr(0, 2));
}

inline string readDate(string prompt) {
    while (true) {
        string date = readText(prompt, 10);
        if (validDate(date)) {
            return date;
        }
        cout << "Ошибка: нужна существующая дата ДД.ММ.ГГГГ, год 1900..2100.\n";
    }
}

inline string readTime(string prompt) {
    while (true) {
        string time = readText(prompt, 5);
        bool valid = time.size() == 5;
        if (valid) {
            valid = time[2] == ':';
            for (int i = 0; i < 5; i++) {
                if (i != 2 && (time[i] < '0' || time[i] > '9')) {
                    valid = false;
                }
            }
        }
        if (valid) {
            int hours = stoi(time.substr(0, 2));
            int minutes = stoi(time.substr(3, 2));
            if (hours <= 23 && minutes <= 59) {
                return time;
            }
        }
        cout << "Ошибка: время должно быть от 00:00 до 23:59.\n";
    }
}

inline string readPhone(string prompt) {
    while (true) {
        string phone = readText(prompt, 16);
        size_t start = 0;
        if (phone[0] == '+') {
            start = 1;
        }
        bool valid = phone.size() - start >= 7 && phone.size() - start <= 15;
        for (size_t i = start; i < phone.size(); i++) {
            if (phone[i] < '0' || phone[i] > '9') {
                valid = false;
            }
        }
        if (valid) {
            return phone;
        }
        cout << "Ошибка: нужно 7..15 цифр; знак + допустим только в начале.\n";
    }
}
```
